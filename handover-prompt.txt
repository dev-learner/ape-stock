import time
from datetime import datetime, timezone, timedelta
from zoneinfo import ZoneInfo
import yfinance as yf
from db import get_db
from data import fetch_quote
import warnings
warnings.filterwarnings("ignore", category=Warning)

UK = ZoneInfo("Europe/London")

def write_snapshot(q):
    if not q or q.get("price") is None:
        return
    minute = datetime.now(UK).minute
    rounded = minute - (minute % 5)
    ts = datetime.now(UK).replace(minute=rounded, second=0, microsecond=0).isoformat()
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        INSERT OR REPLACE INTO snapshots_5min
        (symbol, ts, price, change_pct, volume, avg_volume)
        VALUES (?, ?, ?, ?, ?, ?)
    """, (q["symbol"], ts, q["price"], q["change_pct"], q["volume"], q["avg_volume"]))
    conn.commit()
    conn.close()

def collapse_old_snapshots():
    cutoff = (datetime.now(UK) - timedelta(days=90)).date().isoformat()
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        SELECT symbol, date(ts) as d, price
        FROM snapshots_5min
        WHERE date(ts) < ?
        ORDER BY symbol, ts
    """, (cutoff,))
    rows = cur.fetchall()
    daily = {}
    for r in rows:
        daily[(r["symbol"], r["d"])] = r["price"]
    for (sym, d), close in daily.items():
        cur.execute("""
            INSERT OR REPLACE INTO daily_closes (symbol, date, close)
            VALUES (?, ?, ?)
        """, (sym, d, close))
    cur.execute("DELETE FROM snapshots_5min WHERE date(ts) < ?", (cutoff,))
    conn.commit()
    conn.close()

def catchup_daily_closes():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT MAX(date) FROM daily_closes")
    row = cur.fetchone()
    last_date = row[0] if row and row[0] else None
    today = datetime.now(UK).date()
    if last_date:
        last = datetime.strptime(last_date, "%Y-%m-%d").date()
        if (today - last).days <= 1:
            conn.close()
            print("daily_closes is up to date")
            return
    print(f"Catch-up: daily_closes last date = {last_date}, filling gaps…")
    cur.execute("SELECT DISTINCT symbol FROM index_constituents")
    symbols = [r[0] for r in cur.fetchall()]
    for col, typ in [("open", "REAL"), ("high", "REAL"), ("low", "REAL"), ("volume", "INTEGER")]:
        try:
            cur.execute(f"ALTER TABLE daily_closes ADD COLUMN {col} {typ}")
        except Exception:
            pass
    conn.commit()
    inserted = 0
    for i, sym in enumerate(symbols, 1):
        try:
            hist = yf.Ticker(sym).history(period="10d", interval="1d", auto_adjust=False)
            if hist.empty:
                continue
            for dt, row in hist.iterrows():
                date_str = dt.strftime("%Y-%m-%d")
                if last_date and date_str <= last_date:
                    continue
                o = float(row["Open"]) if row["Open"] == row["Open"] else None
                h = float(row["High"]) if row["High"] == row["High"] else None
                l = float(row["Low"]) if row["Low"] == row["Low"] else None
                c = float(row["Close"]) if row["Close"] == row["Close"] else None
                v = int(row["Volume"]) if row["Volume"] == row["Volume"] else None
                if c is None:
                    continue
                cur.execute("""
                    INSERT OR REPLACE INTO daily_closes
                    (symbol, date, open, high, low, close, volume)
                    VALUES (?, ?, ?, ?, ?, ?, ?)
                """, (sym, date_str, o, h, l, c, v))
                inserted += 1
            if i % 50 == 0:
                conn.commit()
            time.sleep(0.2)
        except Exception as e:
            print(f"  catch-up {sym}: {e}")
    conn.commit()
    conn.close()
    print(f"Catch-up complete — inserted {inserted} rows")

def background_updater():
    last_collapse = datetime.now(UK).date()
    while True:
        try:
            conn = get_db()
            cur = conn.cursor()
            cur.execute("SELECT DISTINCT symbol FROM index_constituents")
            symbols = [row[0] for row in cur.fetchall()]
            conn.close()
            print(f"Background update + snapshots for {len(symbols)} symbols…")
            for sym in symbols:
                q = fetch_quote(sym)
                if q:
                    conn = get_db()
                    cur = conn.cursor()
                    cur.execute("""
                        INSERT INTO index_quotes
                            (symbol, price, change, change_pct, volume, avg_volume, name, last_update)
                        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                        ON CONFLICT(symbol) DO UPDATE SET
                            price = excluded.price,
                            change = excluded.change,
                            change_pct = excluded.change_pct,
                            volume = excluded.volume,
                            avg_volume = COALESCE(excluded.avg_volume, index_quotes.avg_volume),
                            name = excluded.name,
                            last_update = excluded.last_update
                    """, (
                        q["symbol"], q["price"], q["change"], q["change_pct"],
                        q["volume"], q["avg_volume"], q["name"],
                        datetime.now(timezone.utc).isoformat()
                    ))
                    conn.commit()
                    conn.close()
                    write_snapshot(q)
                time.sleep(1.5)
            today = datetime.now(UK).date()
            if today != last_collapse:
                collapse_old_snapshots()
                last_collapse = today
            print("Background cycle complete")
        except Exception as e:
            print("Background error:", e)
        time.sleep(30)

def update_holdings_marks():
    """Call once per day (or on each background cycle after market close)."""
    from holdings import is_uk_trading_day
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM holdings WHERE status = 'open'")
    opens = cur.fetchall()
    today = datetime.now(UK).date()

    for h in opens:
        entry = datetime.strptime(h["entry_date"], "%Y-%m-%d").date()
        # How many trading days since entry
        trading_days = 0
        d = entry
        while d <= today:
            if is_uk_trading_day(d):
                trading_days += 1
            d += timedelta(days=1)
        day_offset = trading_days - 1  # D1 = 0
        if day_offset < 0 or day_offset > 13:
            continue

        # Current price from index_quotes
        cur.execute("SELECT price FROM index_quotes WHERE symbol = ?", (h["symbol"],))
        row = cur.fetchone()
        if not row or not row["price"]:
            continue
        price = row["price"]
        proceeds = h["shares"] * price
        pnl_net = proceeds - h["notional"]

        cur.execute("""
            INSERT OR REPLACE INTO holdings_marks
            (holding_id, day_offset, mark_date, price, pnl_net)
            VALUES (?, ?, ?, ?, ?)
        """, (h["id"], day_offset, today.isoformat(), price, pnl_net))

        # Auto-sell rules
        target_pct = 3.0  # configurable later
        if pnl_net / h["notional"] * 100 >= target_pct or day_offset >= 13:
            reason = "target" if pnl_net / h["notional"] * 100 >= target_pct else "expiry"
            # Re-use close logic
            from holdings import close_holding_internal
            close_holding_internal(h["id"], price, reason)

    conn.commit()
    conn.close()

import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo
import requests
from bs4 import BeautifulSoup
from db import get_db, get_ftse_state, save_ftse_state
import hashlib
from db import get_constituents_fingerprint, save_constituents_fingerprint

UK = ZoneInfo("Europe/London")
_session = requests.Session()
_session.headers.update({"User-Agent": "Mozilla/5.0"})

INDEX_SOURCES = {
    "100": "https://www.hl.co.uk/shares/stock-market-summary/ftse-100",
    "250": "https://www.hl.co.uk/shares/stock-market-summary/ftse-250",
}

def fetch_hl_index(base_url, pages=4):
    symbols = set()
    for page in range(1, pages + 1):
        url = f"{base_url}?page={page}" if page > 1 else base_url
        try:
            r = _session.get(url, timeout=15)
            r.raise_for_status()
            soup = BeautifulSoup(r.text, "html.parser")
            for row in soup.select("table tr"):
                cells = row.find_all("td")
                if not cells:
                    continue
                raw = cells[0].get_text(strip=True).strip()
                if not raw or raw.upper() in ("EPIC", "« PREVIOUS", "NEXT »", "PREVIOUS", "NEXT"):
                    continue
                if any(c in raw for c in "«»") or len(raw) > 12 or " " in raw:
                    continue
                epic = raw.rstrip(".")
                if epic.endswith(".A"):
                    epic = epic[:-2] + "-A"
                elif epic.endswith(".B"):
                    epic = epic[:-2] + "-B"
                elif epic.endswith(".C"):
                    epic = epic[:-2] + "-C"
                if not epic.upper().endswith(".L"):
                    epic = f"{epic}.L"
                symbols.add(epic)
        except Exception as e:
            print(f"HL scrape failed for {url}:", e)
    return symbols

def refresh_constituents():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("DELETE FROM index_constituents")
    for key, url in INDEX_SOURCES.items():
        symbols = fetch_hl_index(url)
        for sym in symbols:
            cur.execute("INSERT OR REPLACE INTO index_constituents (index_key, symbol) VALUES (?, ?)", (key, sym))
        print(f"Loaded {len(symbols)} symbols for FTSE {key}")
    conn.commit()
    conn.close()

def fetch_quote(symbol):
    if not isinstance(symbol, str):
        return None
    symbol = symbol.strip()
    if any(c in symbol for c in "«»") or "Previous" in symbol or "Next" in symbol:
        return None
    if symbol.endswith("..L"):
        symbol = symbol.replace("..L", ".L")
    if ".A.L" in symbol:
        symbol = symbol.replace(".A.L", "-A.L")
    if ".B.L" in symbol:
        symbol = symbol.replace(".B.L", "-B.L")
    url = f"https://query1.finance.yahoo.com/v8/finance/chart/{symbol}"
    try:
        r = _session.get(url, params={"range": "1d", "interval": "1d"}, timeout=10)
        r.raise_for_status()
        meta = r.json()["chart"]["result"][0]["meta"]
        price = meta.get("regularMarketPrice")
        prev = meta.get("chartPreviousClose") or meta.get("previousClose")
        change = price - prev if price and prev else None
        change_pct = (change / prev * 100) if change and prev else None
        return {
            "symbol": symbol,
            "price": price,
            "change": change,
            "change_pct": change_pct,
            "volume": meta.get("regularMarketVolume"),
            "avg_volume": meta.get("averageDailyVolume3Month") or meta.get("averageDailyVolume10Day"),
            "name": meta.get("shortName") or meta.get("longName") or symbol,
        }
    except Exception as e:
        print(f"quote error {symbol}: {e}")
        return None

def fetch_quotes(symbols):
    results = []
    for sym in symbols:
        q = fetch_quote(sym)
        if q:
            results.append(q)
        time.sleep(0.15)
    return results

def fetch_news(symbol):
    try:
        url = f"https://query1.finance.yahoo.com/v1/finance/search"
        r = _session.get(url, params={"q": symbol, "newsCount": 5}, timeout=10)
        r.raise_for_status()
        items = r.json().get("news", [])[:5]
        result = []
        for it in items:
            result.append({
                "title": it.get("title"),
                "time": it.get("providerPublishTime"),
                "url": it.get("link") or it.get("url")
            })
        return result
    except Exception as e:
        print(f"news error {symbol}: {e}")
        return []

def fetch_ftse():
    try:
        r = _session.get(
            "https://query1.finance.yahoo.com/v8/finance/chart/%5EFTSE",
            params={"range": "5d", "interval": "1d"},
            timeout=10
        )
        r.raise_for_status()
        meta = r.json()["chart"]["result"][0]["meta"]
        prev_close = meta.get("chartPreviousClose") or meta.get("previousClose")
        current = meta.get("regularMarketPrice")

        now_uk = datetime.now(UK)
        open_t = now_uk.replace(hour=8, minute=0, second=0, microsecond=0)
        close_t = now_uk.replace(hour=16, minute=30, second=0, microsecond=0)
        is_weekday = now_uk.weekday() < 5
        is_open = is_weekday and open_t <= now_uk < close_t

        state = get_ftse_state()
        previous_was_up = state["previous_was_up"]

        if prev_close is not None and state["previous_close"] != prev_close:
            if state["previous_close"] is not None and current is not None:
                previous_was_up = current >= state["previous_close"]
                save_ftse_state(prev_close, previous_was_up)
            else:
                save_ftse_state(prev_close, None)

        return {
            "previous": prev_close,
            "current": current,
            "is_open": is_open,
            "previous_was_up": bool(previous_was_up) if previous_was_up is not None else None,
            "server_time": now_uk.isoformat()
        }
    except Exception as e:
        print("FTSE fetch error:", e)
        return {"previous": None, "current": None, "is_open": False, "previous_was_up": None, "server_time": None}

def _page1_fingerprint(url):
    """Fast fingerprint from the first page only."""
    try:
        r = _session.get(url, timeout=10)
        r.raise_for_status()
        soup = BeautifulSoup(r.text, "html.parser")
        epics = []
        for row in soup.select("table tr"):
            cells = row.find_all("td")
            if not cells:
                continue
            raw = cells[0].get_text(strip=True).strip()
            if raw and len(raw) <= 12 and " " not in raw and "«" not in raw:
                epics.append(raw)
        epics.sort()
        return hashlib.md5("|".join(epics).encode()).hexdigest()
    except Exception as e:
        print("Fingerprint failed:", e)
        return None

def refresh_constituents_if_needed():
    """Only full-scrape when the page-1 fingerprint has changed."""
    fp100 = _page1_fingerprint(INDEX_SOURCES["100"])
    fp250 = _page1_fingerprint(INDEX_SOURCES["250"])
    if not fp100 or not fp250:
        print("Fingerprint check failed — skipping refresh")
        return

    current_fp = f"{fp100}:{fp250}"
    stored_fp = get_constituents_fingerprint()

    if current_fp == stored_fp:
        print("Constituents unchanged — skipping full scrape")
        return

    print("Constituents changed — running full refresh")
    refresh_constituents()          # existing full scrape
    save_constituents_fingerprint(current_fp)

# db.py
import sqlite3
from pathlib import Path
import datetime
from datetime import datetime, timezone

DB_PATH = "ape_stock.db"

def get_db():
    conn = sqlite3.connect(DB_PATH, timeout=30)
    conn.execute("PRAGMA journal_mode=WAL;")
    conn.execute("PRAGMA synchronous=NORMAL;")
    conn.execute("PRAGMA busy_timeout=30000;")
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS index_constituents (
            index_key TEXT,
            symbol TEXT,
            PRIMARY KEY (index_key, symbol)
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS index_quotes (
            symbol TEXT PRIMARY KEY,
            price REAL,
            change REAL,
            change_pct REAL,
            volume INTEGER,
            avg_volume INTEGER,
            name TEXT,
            last_update TEXT
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS watchlist (
            symbol TEXT PRIMARY KEY,
            added TEXT,
            added_price REAL,
            news_count INTEGER DEFAULT 0,
            news_snippets TEXT,
            news_checked_at TEXT
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS snapshots_5min (
            symbol TEXT,
            ts TEXT,
            price REAL,
            change_pct REAL,
            volume INTEGER,
            avg_volume INTEGER,
            PRIMARY KEY (symbol, ts)
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS daily_closes (
            symbol TEXT,
            date TEXT,
            close REAL,
            PRIMARY KEY (symbol, date)
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS ftse_state (
            id INTEGER PRIMARY KEY CHECK (id = 1),
            previous_close REAL,
            previous_was_up INTEGER
        )
    """)
    cur.execute("INSERT OR IGNORE INTO ftse_state (id) VALUES (1)")

    cur.execute("""
        CREATE TABLE IF NOT EXISTS constituents_meta (
            id INTEGER PRIMARY KEY CHECK (id = 1),
            fingerprint TEXT,
            updated TEXT
        )
    """)
    cur.execute("INSERT OR IGNORE INTO constituents_meta (id) VALUES (1)")

    cur.execute("""
        CREATE TABLE IF NOT EXISTS holdings (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            symbol TEXT NOT NULL,
            entry_date TEXT NOT NULL,
            entry_price REAL NOT NULL,
            notional REAL NOT NULL,
            stamp_pct REAL NOT NULL,
            shares REAL NOT NULL,
            status TEXT DEFAULT 'open',
            created TEXT
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS holdings_marks (
            holding_id INTEGER,
            day_offset INTEGER,
            mark_date TEXT,
            price REAL,
            pnl_net REAL,
            PRIMARY KEY (holding_id, day_offset)
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS holdings_history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            symbol TEXT,
            entry_date TEXT,
            entry_price REAL,
            exit_date TEXT,
            exit_price REAL,
            notional REAL,
            stamp_pct REAL,
            shares REAL,
            exit_reason TEXT,
            final_pnl_net REAL,
            day_path TEXT,
            closed_at TEXT
        )
    """)
    
    conn.commit()
    conn.close()

def get_ftse_state():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT previous_close, previous_was_up FROM ftse_state WHERE id = 1")
    row = cur.fetchone()
    conn.close()
    return dict(row) if row else {"previous_close": None, "previous_was_up": None}

def save_ftse_state(previous_close, previous_was_up):
    conn = get_db()
    cur = conn.cursor()
    flag = None if previous_was_up is None else (1 if previous_was_up else 0)
    cur.execute("""
        UPDATE ftse_state SET previous_close = ?, previous_was_up = ? WHERE id = 1
    """, (previous_close, flag))
    conn.commit()
    conn.close()

# holdings.py
import json
from datetime import datetime, timedelta
from zoneinfo import ZoneInfo
from flask import Blueprint, request, jsonify
from db import get_db

UK = ZoneInfo("Europe/London")
bp = Blueprint("holdings", __name__)

def is_uk_trading_day(d):
    """Simple Mon-Fri check (bank holidays ignored for now)."""
    return d.weekday() < 5

def close_holding_internal(hid, exit_price, reason):
    """Close a holding and archive it. Used by both the route and background."""
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM holdings WHERE id = ?", (hid,))
    h = cur.fetchone()
    if not h:
        conn.close()
        return None

    proceeds = h["shares"] * exit_price
    pnl_net = proceeds - h["notional"]

    cur.execute("""
        SELECT day_offset, price, pnl_net FROM holdings_marks
        WHERE holding_id = ? ORDER BY day_offset
    """, (hid,))
    path = [dict(r) for r in cur.fetchall()]

    cur.execute("""
        INSERT INTO holdings_history
        (symbol, entry_date, entry_price, exit_date, exit_price, notional, stamp_pct, shares,
         exit_reason, final_pnl_net, day_path, closed_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    """, (
        h["symbol"], h["entry_date"], h["entry_price"],
        datetime.now(UK).date().isoformat(), exit_price,
        h["notional"], h["stamp_pct"], h["shares"],
        reason, pnl_net, json.dumps(path),
        datetime.now(UK).isoformat()
    ))
    cur.execute("DELETE FROM holdings WHERE id = ?", (hid,))
    cur.execute("DELETE FROM holdings_marks WHERE holding_id = ?", (hid,))
    conn.commit()
    conn.close()
    return pnl_net

@bp.route("/holdings/open", methods=["POST"])
def open_holding():
    data = request.get_json(silent=True) or {}
    symbol = data.get("symbol")
    price = data.get("price")
    if not symbol or not price:
        return jsonify({"error": "symbol and price required"}), 400

    notional = float(data.get("notional") or 500)
    stamp_pct = float(data.get("stamp_pct") or 0.5)
    cost = notional * (stamp_pct / 100.0)
    invest = notional - cost
    shares = invest / float(price)

    conn = get_db()
    cur = conn.cursor()
    now = datetime.now(UK).isoformat()
    entry_date = datetime.now(UK).date().isoformat()
    cur.execute("""
        INSERT INTO holdings (symbol, entry_date, entry_price, notional, stamp_pct, shares, status, created)
        VALUES (?, ?, ?, ?, ?, ?, 'open', ?)
    """, (symbol, entry_date, price, notional, stamp_pct, shares, now))
    holding_id = cur.lastrowid

    cur.execute("""
        INSERT INTO holdings_marks (holding_id, day_offset, mark_date, price, pnl_net)
        VALUES (?, 0, ?, ?, 0)
    """, (holding_id, entry_date, price))
    conn.commit()
    conn.close()
    return jsonify({"ok": True, "id": holding_id})

@bp.route("/holdings/list")
def list_holdings():
    conn = get_db()

@bp.route("/rsi/<symbol>")
def rsi(symbol):
    """Return RSI(7) from daily closes."""
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        SELECT close FROM daily_closes
        WHERE symbol = ? AND close IS NOT NULL
        ORDER BY date DESC LIMIT 30
    """, (symbol,))
    closes = [r[0] for r in cur.fetchall()][::-1]  # oldest first
    conn.close()
    if len(closes) < 8:
        return jsonify({"rsi7": None})

    def calc_rsi(prices, period=7):
        if len(prices) < period + 1:
            return None
        deltas = [prices[i] - prices[i-1] for i in range(1, len(prices))]
        gains = [d if d > 0 else 0 for d in deltas]
        losses = [-d if d < 0 else 0 for d in deltas]
        avg_gain = sum(gains[:period]) / period
        avg_loss = sum(losses[:period]) / period
        for i in range(period, len(deltas)):
            avg_gain = (avg_gain * (period-1) + gains[i]) / period
            avg_loss = (avg_loss * (period-1) + losses[i]) / period
        if avg_loss == 0:
            return 100.0
        rs = avg_gain / avg_loss
        return 100 - (100 / (1 + rs))

    return jsonify({"rsi7": round(calc_rsi(closes, 7), 1)})

import subprocess
import sys
from pathlib import Path
from flask import Blueprint, request, jsonify

SCRIPTS_DIR = Path(__file__).parent / "scripts"
bp = Blueprint("scripts", __name__)

@bp.route("/scripts/list")
def scripts_list():
    if not SCRIPTS_DIR.exists():
        return jsonify([])
    files = sorted([f.name for f in SCRIPTS_DIR.glob("*.py")])
    return jsonify(files)

@bp.route("/scripts/run", methods=["POST"])
def scripts_run():
    data = request.get_json(silent=True) or {}
    name = data.get("script", "").strip()
    if not name or not name.endswith(".py") or "/" in name or "\\" in name:
        return jsonify({"error": "Invalid script name"}), 400
    script_path = SCRIPTS_DIR / name
    if not script_path.exists():
        return jsonify({"error": "Script not found"}), 404
    try:
        result = subprocess.run(
            [sys.executable, str(script_path)],
            capture_output=True,
            text=True,
            timeout=180,
            cwd=str(SCRIPTS_DIR),
            encoding="utf-8",
            errors="replace"
        )
        return jsonify({
            "stdout": result.stdout,
            "stderr": result.stderr,
            "returncode": result.returncode
        })
    except subprocess.TimeoutExpired:
        return jsonify({"error": "Script timed out"}), 500
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@bp.route("/scripts/meta")
def scripts_meta():
    """Return DESC and RUN comments from a script."""
    name = request.args.get("name", "").strip()
    if not name or not name.endswith(".py") or "/" in name or "\\" in name:
        return jsonify({"name": name, "desc": "", "run": ""})

    path = SCRIPTS_DIR / name
    if not path.exists():
        return jsonify({"name": name, "desc": "", "run": ""})

    desc = ""
    run = ""
    try:
        text = path.read_text(encoding="utf-8", errors="replace")
        for line in text.splitlines()[:30]:
            line = line.strip()
            if line.startswith("# DESC:"):
                desc = line[7:].strip()
            elif line.startswith("# RUN:"):
                run = line[6:].strip()
            if desc and run:
                break
    except Exception:
        pass

    return jsonify({"name": name, "desc": desc, "run": run})

function formatNumber(value, type = "price") {
    if (value == null || isNaN(value)) return "";
    const abs = Math.abs(value);
    if (type === "volume") return value.toLocaleString();
    let decimals = 2;
    if (abs === 0) decimals = 2;
    else if (abs < 0.0001) decimals = 6;
    else if (abs < 0.01) decimals = 4;
    else if (abs < 1) decimals = 3;
    return value.toLocaleString(undefined, {
        minimumFractionDigits: decimals,
        maximumFractionDigits: decimals
    });
}

function makeTableSortable(table, options = {}) {
    if (!table) return;
    const getVal = options.getValue || ((cell) => cell.textContent.trim());
    let sortKey = null;
    let sortDir = 1;

    table.querySelectorAll("th[data-key]").forEach(th => {
        th.style.cursor = "pointer";
        th.addEventListener("click", () => {
            const key = th.dataset.key;
            if (sortKey === key) sortDir *= -1;
            else { sortKey = key; sortDir = 1; }

            const tbody = table.querySelector("tbody");
            const rows = [...tbody.querySelectorAll("tr")];
            const colIndex = th.cellIndex;

            rows.sort((a, b) => {
                // Skip details rows (they have fewer cells)
                if (a.classList.contains("details-row") || b.classList.contains("details-row")) return 0;

                const cellA = a.children[colIndex];
                const cellB = b.children[colIndex];
                if (!cellA || !cellB) return 0;

                let va = getVal(cellA);
                let vb = getVal(cellB);
                const na = parseFloat(String(va).replace(/[^0-9.-]/g, ""));
                const nb = parseFloat(String(vb).replace(/[^0-9.-]/g, ""));
                if (!isNaN(na) && !isNaN(nb)) return (na - nb) * sortDir;
                return String(va).localeCompare(String(vb)) * sortDir;
            });

            // Keep details rows attached to their main row
            const details = [...tbody.querySelectorAll("tr.details-row")];
            rows.forEach(r => {
                if (r.classList.contains("details-row")) return; // already handled
                tbody.appendChild(r);
                const sym = r.dataset.symbol;
                const det = details.find(d => d.previousElementSibling === r || d.dataset.symbol === sym);
                if (det) tbody.appendChild(det);
            });

            // Update sort arrows WITHOUT destroying child elements
            table.querySelectorAll("th[data-key]").forEach(h => {
                // Remove only the arrow text node, leave any inputs alone
                [...h.childNodes].forEach(node => {
                    if (node.nodeType === Node.TEXT_NODE) {
                        node.textContent = node.textContent.replace(/ ↑| ↓/, "");
                    }
                });
            });
            // Append arrow as a text node
            th.appendChild(document.createTextNode(sortDir === 1 ? " ↑" : " ↓"));

            if (options.onSort) options.onSort(key, sortDir);
        });
    });
}

function drawSparkline(canvas, values, style = "line") {
    if (!canvas || !values || values.length < 2) return;
    const ctx = canvas.getContext("2d");
    const w = canvas.width = canvas.offsetWidth || 80;
    const h = canvas.height = canvas.offsetHeight || 24;
    ctx.clearRect(0, 0, w, h);
    const min = Math.min(...values);
    const max = Math.max(...values);
    const range = max - min || 1;
    const step = w / (values.length - 1);

    const points = values.map((v, i) => ({
        x: i * step,
        y: h - ((v - min) / range) * (h - 4) - 2
    }));

    ctx.lineWidth = 1.5;
    if (style === "bars") {
        points.forEach((p, i) => {
            const up = i === 0 || values[i] >= values[i - 1];
            ctx.fillStyle = up ? "#4ade80" : "#f87171";
            const barH = Math.abs(p.y - h / 2);
            ctx.fillRect(p.x - 1, Math.min(p.y, h / 2), 2, barH || 1);
        });
        return;
    }
    if (style === "winloss") {
        points.forEach((p, i) => {
            if (i === 0) return;
            ctx.fillStyle = values[i] >= values[i - 1] ? "#4ade80" : "#f87171";
            ctx.fillRect(p.x - 1, h / 2 - 3, 2, 6);
        });
        return;
    }

    ctx.beginPath();
    ctx.moveTo(points[0].x, points[0].y);
    points.forEach(p => ctx.lineTo(p.x, p.y));
    if (style === "gradient") {
        const grad = ctx.createLinearGradient(0, 0, w, 0);
        grad.addColorStop(0, "#4ade80");
        grad.addColorStop(1, "#f87171");
        ctx.strokeStyle = grad;
    } else {
        ctx.strokeStyle = values[values.length - 1] >= values[0] ? "#4ade80" : "#f87171";
    }
    ctx.stroke();

    if (style === "area") {
        ctx.lineTo(points[points.length - 1].x, h);
        ctx.lineTo(points[0].x, h);
        ctx.closePath();
        ctx.fillStyle = "rgba(74, 222, 128, 0.15)";
        ctx.fill();
    }
}

function nextOpenDescription(nowIso) {
    if (!nowIso) return null;
    const ukNow = new Date(nowIso);
    const day = ukNow.getDay();
    const openToday = new Date(ukNow);
    openToday.setHours(8, 0, 0, 0);
    const closeToday = new Date(ukNow);
    closeToday.setHours(16, 30, 0, 0);

    let target = new Date(openToday);
    let label = "";

    if (day === 0) {                       // Sunday
        label = "Opens [tomorrow in]:";
        target.setDate(target.getDate() + 1);
    } else if (day === 6) {                // Saturday
        label = "Opens [on Monday in]:";
        target.setDate(target.getDate() + 2);
    } else if (ukNow >= closeToday) {      // after close
        label = "Opens [tomorrow in]:";
        target.setDate(target.getDate() + 1);
        if (target.getDay() === 6) {
            label = "Opens [on Monday in]:";
            target.setDate(target.getDate() + 2);
        } else if (target.getDay() === 0) {
            label = "Opens [on Monday in]:";
            target.setDate(target.getDate() + 1);
        }
    } else if (ukNow < openToday) {        // before open
        label = "Opens [today in]:";
    } else {
        return null;                       // market is open
    }

    // Safety: if target is already past, push to next weekday open
    if (target <= ukNow) {
        target.setDate(target.getDate() + 1);
        while (target.getDay() === 0 || target.getDay() === 6) {
            target.setDate(target.getDate() + 1);
        }
        label = "Opens [next in]:";
    }

    return { label, target };
}

function startFtseCountdown(target, labelEl) {
    if (ftseCountdownTimer) clearInterval(ftseCountdownTimer);
    function tick() {
        const diff = target - new Date();
        if (diff <= 0) {
            labelEl.textContent = "Open";
            clearInterval(ftseCountdownTimer);
            return;
        }
        const h = Math.floor(diff / 3600000);
        const m = Math.floor((diff % 3600000) / 60000);
        const s = Math.floor((diff % 60000) / 1000);
        labelEl.textContent = `${String(h).padStart(2,"0")}:${String(m).padStart(2,"0")}:${String(s).padStart(2,"0")}`;
    }
    tick();
    ftseCountdownTimer = setInterval(tick, 1000);
}

// Re-apply a saved sort only if the visual order has actually changed.
// Avoids unnecessary DOM moves (and therefore flicker) on price ticks.
function reapplySortIfChanged(table, key, dir) {
    if (!table || !key) return;
    const th = table.querySelector(`th[data-key="${key}"]`);
    if (!th) return;

    const tbody = table.querySelector("tbody");
    if (!tbody) return;
    const rows = [...tbody.querySelectorAll("tr")];
    if (rows.length < 2) return;

    const colIndex = th.cellIndex;

    const sorted = [...rows].sort((a, b) => {
        let va = a.children[colIndex].textContent.trim();
        let vb = b.children[colIndex].textContent.trim();
        const na = parseFloat(String(va).replace(/[^0-9.-]/g, ""));
        const nb = parseFloat(String(vb).replace(/[^0-9.-]/g, ""));
        if (!isNaN(na) && !isNaN(nb)) return (na - nb) * dir;
        return String(va).localeCompare(String(vb)) * dir;
    });

    // Already in the correct order?
    let same = true;
    for (let i = 0; i < rows.length; i++) {
        if (rows[i] !== sorted[i]) {
            same = false;
            break;
        }
    }
    if (same) return;

    // Order changed — move rows
    sorted.forEach(r => tbody.appendChild(r));

    // Update sort indicator
    table.querySelectorAll("th[data-key]").forEach(h => {
        [...h.childNodes].forEach(node => {
            if (node.nodeType === Node.TEXT_NODE) {
                node.textContent = node.textContent.replace(/ ↑| ↓/, "");
            }
        });
    });
    th.appendChild(document.createTextNode(dir === 1 ? " ↑" : " ↓"));
}

            const selected = document.querySelector('input[name="script-choice"]:checked');
            if (!selected) {
                alert("Select a script first");
                return;
            }

            const output = document.getElementById("script-output");
            const name = selected.value;

            // Check run mode first
            try {
                const metaRes = await fetch(`/scripts/meta?name=${encodeURIComponent(name)}`);
                const meta = await metaRes.json();
                if (meta.run && meta.run.toUpperCase() === "VENV") {

            }

                const res = await fetch("/scripts/run", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({ script: name })
                });
                const data = await res.json();
                if (data.error) {
                    output.textContent = "Error: " + data.error;
                } else {
                    output.textContent = (data.stdout || "") + (data.stderr ? "\n--- stderr ---\n" + data.stderr : "");
                }
            } catch (e) {
                output.textContent = "Request failed: " + e;
            }
        });

                    drawSparkline(canvas, prices, style);
                } catch (e) {
                    drawSparkline(canvas, [1, 1.02, 0.98, 1.01, 1.03], style);
                }

                const addedEl = document.querySelector(`.added-price[data-symbol="${sym}"]`);
                const dateEl = document.querySelector(`.added-date[data-symbol="${sym}"]`);
                const sinceEl = document.querySelector(`.since-added[data-symbol="${sym}"]`);

                if (row.added) {
                    const d = new Date(row.added);
                    const dd = String(d.getDate()).padStart(2, "0");
                    const mm = String(d.getMonth() + 1).padStart(2, "0");
                    const yyyy = d.getFullYear();
                    if (dateEl) dateEl.textContent = `${dd}/${mm}/${yyyy}`;
                }
                if (row.added_price != null && addedEl) {
                    addedEl.textContent = formatNumber(row.added_price, "price");
                }

                const added = row.added_price;
                const current = row.price;
                if (added && current && sinceEl) {
                    const diff = current - added;
                    const pct = (diff / added * 100);
                    const diffText = (diff >= 0 ? "+" : "") + formatNumber(diff, "price");
                    const pctText = (pct >= 0 ? "+" : "") + pct.toFixed(1) + "%";
                    sinceEl.textContent = `${diffText} (${pctText})`;
                    sinceEl.className = "since-added " + (diff >= 0 ? "positive" : "negative");
                }
            });

            if (window.AppState.wlSort && window.AppState.wlSort.key) {
                reapplySortIfChanged(table, window.AppState.wlSort.key, window.AppState.wlSort.dir);
            }

            this.updateRowNumbersAndPoll();
        });
    },

    updateRowNumbersAndPoll() {
        const table = document.getElementById("wl-table");
        if (!table) return;
        const rows = [...table.querySelectorAll("tbody tr")];

        rows.forEach((tr, idx) => {
            const numCell = tr.querySelector(".row-num");
            if (numCell) numCell.textContent = idx + 1;
        });

        if (!window.AppState.pollActive || !window.AppState.pollPrices) {
            rows.forEach(tr => {
                const cell = tr.querySelector(".poll-cell");
                if (cell) cell.innerHTML = "";
            });
            return;
        }

        if (!window.AppState.pollPendingNew) {
            window.AppState.pollPendingNew = new Set();
        }

        const pollPrices = window.AppState.pollPrices;
        const currentPrices = {};
        let anyExistingMoved = false;

        rows.forEach(tr => {
            const sym = tr.dataset.symbol;
            const priceCell = tr.querySelector("[data-field=price]");
            const current = priceCell ? parseFloat(priceCell.textContent.replace(/[^0-9.-]/g, "")) : NaN;
            currentPrices[sym] = current;

            if (!(sym in pollPrices)) {
                window.AppState.pollPendingNew.add(sym);
            } else if (!isNaN(current) && current !== pollPrices[sym]) {
                anyExistingMoved = true;
            }
        });

        // Only promote stars on a *subsequent* render after an existing price has moved
        if (anyExistingMoved && window.AppState.pollPendingNew.size) {
            window.AppState.pollPendingNew.forEach(sym => {
                if (!(sym in pollPrices) && !isNaN(currentPrices[sym])) {
                    pollPrices[sym] = currentPrices[sym];
                }
            });
            window.AppState.pollPendingNew.clear();
            localStorage.setItem("pollPrices", JSON.stringify(pollPrices));
        }

        rows.forEach(tr => {
            const cell = tr.querySelector(".poll-cell");
            if (!cell) return;
            const sym = tr.dataset.symbol;
            const current = currentPrices[sym];

            if (window.AppState.pollPendingNew.has(sym)) {
                cell.innerHTML = this.pollIcon("new");
                return;
            }

            if (!(sym in pollPrices) || isNaN(current) || isNaN(pollPrices[sym])) {
                cell.innerHTML = this.pollIcon("stable");
                return;
            }

            const base = pollPrices[sym];
            if (current > base) cell.innerHTML = this.pollIcon("up");
            else if (current < base) cell.innerHTML = this.pollIcon("down");
            else cell.innerHTML = this.pollIcon("stable");
        });
    },

    pollIcon(type) {
        if (type === "up") {
            return `<svg width="14" height="14" viewBox="0 0 14 14" style="vertical-align:middle"><polygon points="7,2 12,10 2,10" fill="#4ade80"/></svg>`;
        }
        if (type === "down") {
            return `<svg width="14" height="14" viewBox="0 0 14 14" style="vertical-align:middle"><polygon points="7,12 12,4 2,4" fill="#f87171"/></svg>`;
        }
        if (type === "new") {
            return `<svg width="14" height="14" viewBox="0 0 14 14" style="vertical-align:middle"><polygon points="7,1 9,5 13,5.5 10,8.5 11,13 7,10.5 3,13 4,8.5 1,5.5 5,5" fill="#e67e22"/></svg>`;
        }
        return `<svg width="12" height="12" viewBox="0 0 12 12" style="vertical-align:middle"><circle cx="6" cy="6" r="5" fill="#4ade80"/></svg>`;
    }
};

        const expandOn = side === "risers" ? window.AppState.expandSelectedRisers : window.AppState.expandSelectedFallers;

        let html = `<table class="data-table"><thead><tr>

            const hitCls = hitPct >= 75 ? "positive" : hitPct >= 50 ? "" : "negative";
            const hitTitle = reasons.length ? reasons.join("\n") : "No rules matched";

    try:
        cur.execute(f"ALTER TABLE daily_closes ADD COLUMN {col} {typ}")
        print(f"Added {col} column")

        pass
conn.commit()

    try:
        hist = yf.Ticker(sym).history(period="1y", interval="1d", auto_adjust=False)

        for dt, row in hist.iterrows():

            o = float(row["Open"]) if row["Open"] == row["Open"] else None

# scripts/create_handover_prompt.py
# DESC: Generate handover-prompt.txt with full source tree and file contents
# RUN: Tool2

from pathlib import Path

ROOT = Path(__file__).parent.parent
OUT = ROOT / "handover-prompt.txt"

# Everything we want (explicit so venv, __pycache__, db, etc. are excluded)
INCLUDE_DIRS = ["templates", "static", "scripts"]
INCLUDE_ROOT_FILES = [
    "app.py",
    "background.py",
    "data.py",
    "db.py",
    "holdings.py",
    "routes.py",
    "scripts_runner.py",
    "requirements.txt",
    "run.bat",
    "set-env.bat",
]

def should_include(path: Path) -> bool:
    if path.name.startswith(".") or path.name == "__pycache__":
        return False
    if path.suffix in {".pyc", ".db", ".txt"} and path.name != "requirements.txt":
        return False
    return True

def build_tree():
    lines = ["ape-stock/"]
    # root files
    for name in sorted(INCLUDE_ROOT_FILES):
        p = ROOT / name
        if p.exists():
            lines.append(f"├── {name}")
    # directories
    for dname in INCLUDE_DIRS:
        d = ROOT / dname
        if not d.exists():
            continue
        lines.append(f"├── {dname}/")
        for f in sorted(d.rglob("*")):
            if f.is_file() and should_include(f):
                rel = f.relative_to(ROOT)
                indent = "│   " * (len(rel.parts) - 1)
                lines.append(f"{indent}├── {rel.name}")
    return "\n".join(lines)

def file_blocks():
    blocks = []
    # root files
    for name in INCLUDE_ROOT_FILES:
        p = ROOT / name
        if p.exists() and should_include(p):
            content = p.read_text(encoding="utf-8", errors="replace")
            blocks.append(f"===== FILE: {name} =====\n{content}\n")
    # directory files
    for dname in INCLUDE_DIRS:
        d = ROOT / dname
        if not d.exists():
            continue
        for f in sorted(d.rglob("*")):
            if f.is_file() and should_include(f):
                rel = f.relative_to(ROOT).as_posix()
                content = f.read_text(encoding="utf-8", errors="replace")
                blocks.append(f"===== FILE: {rel} =====\n{content}\n")
    return "\n".join(blocks)

prompt = f"""You are continuing development of a local Flask application called ape-stock.

Current goals, tech stack, design decisions and todo items live on the Dashboard page of the running app.

===== DIRECTORY STRUCTURE =====
{build_tree()}

===== FILE CONTENTS =====
{file_blocks()}

Continue from this exact state. Prefer small, modular, well-commented changes. Keep the UK-only focus and the shared-core philosophy.
"""

OUT.write_text(prompt, encoding="utf-8")
print(f"Wrote {OUT}")

# scripts/gap_down_volume.py
# DESC: Count high-volume gap-downs (≥5 %) and show samples
# RUN: Tool2

import sqlite3
from pathlib import Path
from collections import defaultdict

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path, timeout=30)
conn.row_factory = sqlite3.Row
cur = conn.cursor()

# Load all daily data ordered by symbol, date
cur.execute("""
    SELECT symbol, date, open, close, volume
    FROM daily_closes
    WHERE open IS NOT NULL AND close IS NOT NULL AND volume IS NOT NULL
    ORDER BY symbol, date
""")
rows = cur.fetchall()

# Group by symbol
by_sym = defaultdict(list)
for r in rows:
    by_sym[r["symbol"]].append(r)

GAP_PCT = 5.0          # open 5% below previous close
VOL_WINDOW = 30        # 30-day average volume

total_gaps = 0
gaps_with_high_vol = 0
examples = []

for sym, data in by_sym.items():
    if len(data) < VOL_WINDOW + 1:
        continue
    for i in range(VOL_WINDOW, len(data)):
        prev_close = data[i-1]["close"]
        today_open = data[i]["open"]
        today_vol = data[i]["volume"]
        if prev_close is None or prev_close == 0:
            continue
        gap_pct = (today_open - prev_close) / prev_close * 100
        if gap_pct <= -GAP_PCT:
            # 30-day average volume up to yesterday
            vols = [data[j]["volume"] for j in range(i-VOL_WINDOW, i) if data[j]["volume"]]
            if not vols:
                continue
            avg_vol = sum(vols) / len(vols)
            total_gaps += 1
            high_vol = today_vol > avg_vol
            if high_vol:
                gaps_with_high_vol += 1
            if len(examples) < 15:
                examples.append((sym, data[i]["date"], round(gap_pct, 2), today_vol, int(avg_vol), high_vol))

print(f"Gap-downs >= {GAP_PCT}% (open vs prev close): {total_gaps}")
print(f"Of which volume > 30-day average: {gaps_with_high_vol}")
if total_gaps:
    print(f"Percentage: {gaps_with_high_vol / total_gaps * 100:.1f}%")
print("\nSample occurrences:")
for e in examples:
    flag = "HIGH VOL" if e[5] else ""
    print(f"  {e[0]:<10} {e[1]}  gap {e[2]:>6.2f}%  vol {e[3]:>10,}  avg {e[4]:>10,}  {flag}")

conn.close()

# scripts/gap_grid_search.py
# DESC: Grid-search gap size, volume window and forward days for recovery stats
# RUN: VENV

import sqlite3
from pathlib import Path
from collections import defaultdict

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path, timeout=30)
conn.row_factory = sqlite3.Row
cur = conn.cursor()

cur.execute("""
    SELECT symbol, date, open, close, volume
    FROM daily_closes
    WHERE open IS NOT NULL AND close IS NOT NULL AND volume IS NOT NULL
    ORDER BY symbol, date
""")
rows = cur.fetchall()

by_sym = defaultdict(list)
for r in rows:
    by_sym[r["symbol"]].append(dict(r))

gap_thresholds = [4, 5, 6, 7, 8]
vol_windows = [20, 30, 50]
forward_days = [1, 3, 5]

print(f"{'Gap%':>5} {'VolWin':>6} {'Fwd':>4} {'Signals':>8} {'Recovered%':>11} {'AvgRet%':>8}")
print("-" * 50)

for gap in gap_thresholds:
    for win in vol_windows:
        for fwd in forward_days:
            signals = 0
            recovered = 0
            total_ret = 0.0
            for data in by_sym.values():
                if len(data) < win + fwd + 1:
                    continue
                for i in range(win, len(data) - fwd):
                    prev_c = data[i-1]["close"]
                    if not prev_c:
                        continue
                    gap_pct = (data[i]["open"] - prev_c) / prev_c * 100
                    if gap_pct > -gap:
                        continue
                    vols = [data[j]["volume"] for j in range(i-win, i)]
                    avg_vol = sum(vols) / len(vols) if vols else 0
                    if not avg_vol or data[i]["volume"] <= avg_vol:
                        continue
                    future_c = data[i+fwd]["close"]
                    if not future_c or not data[i]["open"]:
                        continue
                    ret = (future_c - data[i]["open"]) / data[i]["open"] * 100
                    signals += 1
                    total_ret += ret
                    if ret >= (gap * 0.5):
                        recovered += 1
            rec_pct = (recovered / signals * 100) if signals else 0
            avg_ret = (total_ret / signals) if signals else 0
            print(f"{gap:5} {win:6} {fwd:4} {signals:8} {rec_pct:10.1f}% {avg_ret:8.2f}")

conn.close()

# scripts/nwg_last_14.py
# DESC: Show last ~14 daily closes and day-to-day % change for NWG.L
# RUN: Tool2

import sqlite3
from pathlib import Path

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path, timeout=30)
conn.row_factory = sqlite3.Row
cur = conn.cursor()

cur.execute("""
    SELECT date, open, high, low, close, volume
    FROM daily_closes
    WHERE symbol = 'NWG.L'
    ORDER BY date DESC
    LIMIT 15
""")
rows = cur.fetchall()
rows = list(reversed(rows))   # oldest first

if len(rows) < 2:
    print("Not enough NWG.L data")
else:
    print(f"{'Date':<12} {'Close':>10} {'Chg%':>8} {'Volume':>12}")
    print("-" * 46)
    for i, r in enumerate(rows):
        if i == 0:
            chg = ""
        else:
            prev = rows[i-1]["close"]
            if prev:
                chg = f"{(r['close'] - prev) / prev * 100:+.2f}%"
            else:
                chg = ""
        vol = f"{r['volume']:,}" if r["volume"] else ""
        print(f"{r['date']:<12} {r['close']:10.2f} {chg:>8} {vol:>12}")

conn.close()

# quick_faller_volume.py
# DESC: List current fallers ≤ –5 % and flag those with volume > average
# RUN: Tool2

import sqlite3
import os
from pathlib import Path

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path)
conn.row_factory = sqlite3.Row
cur = conn.cursor()

# Adjust the fall threshold here
FALL_PCT = -5.0

cur.execute("""
    SELECT symbol, name, price, change_pct, volume, avg_volume
    FROM index_quotes
    WHERE change_pct <= ?
      AND avg_volume IS NOT NULL
      AND avg_volume > 0
    ORDER BY change_pct ASC
""", (FALL_PCT,))

rows = cur.fetchall()
print(f"Fallers <= {FALL_PCT}% : {len(rows)}")
print("-" * 70)

high_vol = 0
for r in rows:
    flag = "HIGH VOL" if r["volume"] > r["avg_volume"] else ""
    if flag:
        high_vol += 1
    print(f"{r['symbol']:<10} {r['change_pct']:>7.2f}%  "
          f"vol {r['volume']:>12,}  avg {r['avg_volume']:>12,}  {flag}")

print("-" * 70)
print(f"Of which volume > average volume: {high_vol}")
if rows:
    print(f"Percentage: {high_vol / len(rows) * 100:.1f}%")

conn.close()

# scripts/update_fundamentals.py
# DESC: Fill sector, industry, market cap, PE, yields, dates and other fundamentals
# RUN: VENV

"""
Fill sector, industry, market_cap, avg_volume, PE, dividend yield,
shares outstanding, 52-week high/low, beta, book value, P/B,
ex-dividend date and earnings date.
"""
import sqlite3
import time
from pathlib import Path
from datetime import datetime, timezone
import yfinance as yf
import warnings
warnings.filterwarnings("ignore")

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path, timeout=30)
cur = conn.cursor()

columns = [
    ("sector", "TEXT"),
    ("industry", "TEXT"),
    ("market_cap", "INTEGER"),
    ("avg_volume", "INTEGER"),
    ("pe_trailing", "REAL"),
    ("pe_forward", "REAL"),
    ("dividend_yield", "REAL"),
    ("shares_outstanding", "INTEGER"),
    ("fifty_two_week_high", "REAL"),
    ("fifty_two_week_low", "REAL"),
    ("beta", "REAL"),
    ("book_value", "REAL"),
    ("price_to_book", "REAL"),
    ("ex_dividend_date", "TEXT"),
    ("earnings_date", "TEXT"),
]

for col, typ in columns:
    try:
        cur.execute(f"ALTER TABLE index_quotes ADD COLUMN {col} {typ}")
        print(f"Added column: {col}")
    except sqlite3.OperationalError:
        pass
conn.commit()

def ts_to_date(ts):
    if not ts:
        return None
    try:
        return datetime.fromtimestamp(int(ts), tz=timezone.utc).strftime("%Y-%m-%d")
    except Exception:
        return None

cur.execute("SELECT symbol FROM index_quotes")
symbols = [r[0] for r in cur.fetchall()]
print(f"Updating fundamentals for {len(symbols)} symbols…")

updated = 0
for i, sym in enumerate(symbols, 1):
    try:
        info = yf.Ticker(sym).info or {}

        data = {
            "sector": info.get("sector") or info.get("sectorDisp"),
            "industry": info.get("industry") or info.get("industryDisp"),
            "market_cap": info.get("marketCap"),
            "avg_volume": info.get("averageDailyVolume3Month") or info.get("averageDailyVolume10Day"),
            "pe_trailing": info.get("trailingPE"),
            "pe_forward": info.get("forwardPE"),
            "dividend_yield": info.get("dividendYield"),
            "shares_outstanding": info.get("sharesOutstanding"),
            "fifty_two_week_high": info.get("fiftyTwoWeekHigh"),
            "fifty_two_week_low": info.get("fiftyTwoWeekLow"),
            "beta": info.get("beta"),
            "book_value": info.get("bookValue"),
            "price_to_book": info.get("priceToBook"),
            "ex_dividend_date": ts_to_date(info.get("exDividendDate")),
            "earnings_date": ts_to_date(info.get("earningsTimestamp") or info.get("earningsTimestampStart")),
        }

        set_clause = ", ".join(f"{k} = COALESCE(?, {k})" for k in data)
        values = list(data.values()) + [sym]

        cur.execute(f"""
            UPDATE index_quotes
            SET {set_clause}
            WHERE symbol = ?
        """, values)

        if cur.rowcount:
            updated += 1

        if i % 25 == 0:
            print(f"  …{i}/{len(symbols)}", flush=True)
            conn.commit()
        time.sleep(0.25)
    except Exception as e:
        print(f"  {sym}: {e}")

conn.commit()
conn.close()
print(f"Done. Updated {updated} symbols.")

# scripts/update_sectors.py
# DESC: Early version – fill sector and industry only
# RUN: VENV

"""
Add sector & industry to index_quotes for all FTSE constituents.
"""
import sqlite3
import time
from pathlib import Path
import yfinance as yf
import warnings
warnings.filterwarnings("ignore", category=Warning)

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path, timeout=30)
cur = conn.cursor()

# Add columns if they don't exist
for col in ("sector", "industry"):
    try:
        cur.execute(f"ALTER TABLE index_quotes ADD COLUMN {col} TEXT")
        print(f"Added column: {col}")
    except sqlite3.OperationalError:
        pass  # already exists
conn.commit()

# Get all symbols
cur.execute("SELECT DISTINCT symbol FROM index_constituents")
symbols = [r[0] for r in cur.fetchall()]
print(f"Updating sector/industry for {len(symbols)} symbols…")

updated = 0
for i, sym in enumerate(symbols, 1):
    try:
        info = yf.Ticker(sym).info
        sector = info.get("sector") or info.get("sectorDisp")
        industry = info.get("industry") or info.get("industryDisp")
        if sector or industry:
            cur.execute("""
                UPDATE index_quotes
                SET sector = ?, industry = ?
                WHERE symbol = ?
            """, (sector, industry, sym))
            if cur.rowcount:
                updated += 1
        if i % 25 == 0:
            print(f"  …{i}/{len(symbols)}", flush=True)
            conn.commit()
        time.sleep(0.25)
    except Exception as e:
        print(f"  {sym}: {e}")

conn.commit()
conn.close()
print(f"Done. Updated {updated} symbols.")
