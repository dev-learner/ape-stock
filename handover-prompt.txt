def ensure_screener_metrics():
    """Run screener metrics update if it has not been run since the last trading day."""
    from db import get_db
    UK = ZoneInfo("Europe/London")
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT value FROM app_settings WHERE key = 'screener_metrics_updated'")
    row = cur.fetchone()
    conn.close()

    last = None
    if row and row[0]:
        try:
            last = datetime.fromisoformat(row[0])
        except Exception:
            last = None

    now = datetime.now(UK)
    needs_run = last is None or (now - last) > timedelta(hours=20)

    if needs_run:
        print("Screener metrics stale or missing — updating…")
        script = Path(__file__).parent / "scripts" / "update_screener_metrics.py"
        try:
            subprocess.run([sys.executable, str(script)], check=False, timeout=120)
        except Exception as e:
            print(f"Screener metrics update failed: {e}")
    else:
        print(f"Screener metrics up to date (last={last})")

if __name__ == "__main__":
    init_db()
    catchup_daily_closes()
    ensure_screener_metrics()
    threading.Thread(target=background_updater, daemon=True).start()
    app.run(host="0.0.0.0", port=5050, debug=False)

def write_snapshot(q):
    if not q or q.get("price") is None:
        return
    minute = datetime.now(UK).minute
    rounded = minute - (minute % 5)
    ts = datetime.now(UK).replace(minute=rounded, second=0, microsecond=0).isoformat()
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        INSERT OR REPLACE INTO snapshots_5min
        (symbol, ts, price, change_pct, volume, avg_volume)
        VALUES (?, ?, ?, ?, ?, ?)
    """, (q["symbol"], ts, q["price"], q["change_pct"], q["volume"], q["avg_volume"]))
    conn.commit()
    conn.close()

def collapse_old_snapshots():
    cutoff = (datetime.now(UK) - timedelta(days=90)).date().isoformat()
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        SELECT symbol, date(ts) as d, price
        FROM snapshots_5min
        WHERE date(ts) < ?
        ORDER BY symbol, ts
    """, (cutoff,))
    rows = cur.fetchall()
    daily = {}
    for r in rows:
        daily[(r["symbol"], r["d"])] = r["price"]
    for (sym, d), close in daily.items():
        cur.execute("""
            INSERT OR REPLACE INTO daily_closes (symbol, date, close)
            VALUES (?, ?, ?)
        """, (sym, d, close))
    cur.execute("DELETE FROM snapshots_5min WHERE date(ts) < ?", (cutoff,))
    conn.commit()
    conn.close()

def catchup_daily_closes():
    """Fill any missing daily closes. Skips when the market is closed and data is already current."""
    now = datetime.now(UK)
    is_weekday = now.weekday() < 5
    market_open = is_weekday and 8 <= now.hour < 17

    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT MAX(date) FROM daily_closes")
    row = cur.fetchone()
    last_date_str = row[0] if row and row[0] else None
    conn.close()

    def last_trading_day(d):
        """Return the most recent Mon-Fri on or before d."""
        while d.weekday() >= 5:          # Sat/Sun
            d -= timedelta(days=1)
        return d

    today = now.date()
    expected = last_trading_day(today if market_open or now.hour >= 17 else today - timedelta(days=1))

    if last_date_str:
        last_dt = datetime.fromisoformat(last_date_str).date()
        if not market_open and last_dt >= expected:
            print(f"Catch-up: market closed, daily_closes already current (last={last_date_str}) — skipping")
            return
        print(f"Catch-up: daily_closes last date = {last_date_str}, filling gaps…")
    else:
        print("Catch-up: no daily_closes yet — running full fill")

    # --- fill logic ---
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT DISTINCT symbol FROM index_constituents")
    symbols = [r[0] for r in cur.fetchall()]
    conn.close()

    if not symbols:
        print("Catch-up: no constituents found")
        return

    end = now.date()
    if last_date_str:
        start = datetime.fromisoformat(last_date_str).date() + timedelta(days=1)
    else:
        start = end - timedelta(days=30)

    if start > end:
        print("Catch-up: nothing to fill")
        return

    print(f"Catch-up: filling {len(symbols)} symbols from {start} to {end}")
    filled = 0
    for i, sym in enumerate(symbols, 1):
        try:
            ticker = yf.Ticker(sym)
            hist = ticker.history(start=start.isoformat(), end=(end + timedelta(days=1)).isoformat(), interval="1d")
            if hist.empty:
                continue
            conn = get_db()
            cur = conn.cursor()
            for idx, row in hist.iterrows():
                d = idx.date().isoformat()
                close = row["Close"]
                vol = row["Volume"]
                if close and close == close:  # not NaN
                    cur.execute("""
                        INSERT OR REPLACE INTO daily_closes (symbol, date, close, volume)
                        VALUES (?, ?, ?, ?)
                    """, (sym, d, float(close), int(vol) if vol == vol else None))
                    filled += 1
            conn.commit()
            conn.close()
            if i % 25 == 0:
                print(f"  …{i}/{len(symbols)}", flush=True)
            time.sleep(0.2)
        except Exception as e:
            print(f"  {sym}: {e}")
    print(f"Catch-up: done. Wrote {filled} daily bars.")

def background_updater():
    last_collapse = datetime.now(UK).date()
    last_holdings_daily = None
    while True:
        try:
            conn = get_db()
            cur = conn.cursor()
            cur.execute("SELECT DISTINCT symbol FROM index_constituents")
            symbols = [row[0] for row in cur.fetchall()]
            conn.close()
            print(f"Background update + snapshots for {len(symbols)} symbols…")
            for sym in symbols:
                q = fetch_quote(sym)
                if q:
                    conn = get_db()
                    cur = conn.cursor()
                    cur.execute("""
                        INSERT INTO index_quotes
                            (symbol, price, change, change_pct, volume, avg_volume, name, last_update)
                        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                        ON CONFLICT(symbol) DO UPDATE SET
                            price = excluded.price,
                            change = excluded.change,
                            change_pct = excluded.change_pct,
                            volume = excluded.volume,
                            avg_volume = COALESCE(excluded.avg_volume, index_quotes.avg_volume),
                            name = excluded.name,
                            last_update = excluded.last_update
                    """, (
                        q["symbol"], q["price"], q["change"], q["change_pct"],
                        q["volume"], q["avg_volume"], q["name"],
                        datetime.now(timezone.utc).isoformat()
                    ))
                    conn.commit()
                    conn.close()
                    write_snapshot(q)
                time.sleep(1.5)
            today = datetime.now(UK).date()
            if today != last_collapse:
                collapse_old_snapshots()
                last_collapse = today
            if last_holdings_daily != today:
                update_holdings_marks()
                last_holdings_daily = today
            print("Background cycle complete")
        except Exception as e:
            print("Background error:", e)
        time.sleep(30)

def update_holdings_marks():
    """Once-per-day: advance day offsets, freeze prior marks, apply auto-sell rules."""
    from holdings import is_uk_trading_day, close_holding_internal
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM holdings WHERE status = 'open'")
    opens = cur.fetchall()
    today = datetime.now(UK).date()
    target_pct = float(get_setting("tracker_close_target", "3"))

    for h in opens:
        entry = datetime.strptime(h["entry_date"], "%Y-%m-%d").date()
        trading_days = 0
        d = entry
        while d <= today:
            if is_uk_trading_day(d):
                trading_days += 1
            d += timedelta(days=1)
        day_offset = trading_days - 1
        if day_offset < 0 or day_offset > 13:
            continue

        cur.execute("SELECT price FROM index_quotes WHERE symbol = ?", (h["symbol"],))
        row = cur.fetchone()
        if not row or not row["price"]:
            continue
        price = row["price"]
        proceeds = h["shares"] * price
        pnl_net = proceeds - h["notional"]

        cur.execute("""
            INSERT OR REPLACE INTO holdings_marks
            (holding_id, day_offset, mark_date, price, pnl_net)
            VALUES (?, ?, ?, ?, ?)
        """, (h["id"], day_offset, today.isoformat(), price, pnl_net))

        if pnl_net / h["notional"] * 100 >= target_pct or day_offset >= 13:
            reason = "target" if pnl_net / h["notional"] * 100 >= target_pct else "expiry"
            close_holding_internal(h["id"], price, reason)

    conn.commit()
    conn.close()

import time
from datetime import datetime, timezone

UK = ZoneInfo("Europe/London")
_session = requests.Session()
_session.headers.update({"User-Agent": "Mozilla/5.0"})

INDEX_SOURCES = {
    "100": "https://www.hl.co.uk/shares/stock-market-summary/ftse-100",
    "250": "https://www.hl.co.uk/shares/stock-market-summary/ftse-250",
}

def fetch_hl_index(base_url, pages=4):
    symbols = set()
    for page in range(1, pages + 1):
        url = f"{base_url}?page={page}" if page > 1 else base_url
        try:
            r = _session.get(url, timeout=15)
            r.raise_for_status()
            soup = BeautifulSoup(r.text, "html.parser")
            for row in soup.select("table tr"):
                cells = row.find_all("td")
                if not cells:
                    continue
                raw = cells[0].get_text(strip=True).strip()
                if not raw or raw.upper() in ("EPIC", "« PREVIOUS", "NEXT »", "PREVIOUS", "NEXT"):
                    continue
                if any(c in raw for c in "«»") or len(raw) > 12 or " " in raw:
                    continue
                epic = raw.rstrip(".")
                if epic.endswith(".A"):
                    epic = epic[:-2] + "-A"
                elif epic.endswith(".B"):
                    epic = epic[:-2] + "-B"
                elif epic.endswith(".C"):
                    epic = epic[:-2] + "-C"
                if not epic.upper().endswith(".L"):
                    epic = f"{epic}.L"
                symbols.add(epic)
        except Exception as e:
            print(f"HL scrape failed for {url}:", e)
    return symbols

def refresh_constituents():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("DELETE FROM index_constituents")
    for key, url in INDEX_SOURCES.items():
        symbols = fetch_hl_index(url)
        for sym in symbols:
            cur.execute("INSERT OR REPLACE INTO index_constituents (index_key, symbol) VALUES (?, ?)", (key, sym))
        print(f"Loaded {len(symbols)} symbols for FTSE {key}")
    conn.commit()
    conn.close()

def fetch_quote(symbol):
    if not isinstance(symbol, str):
        return None
    symbol = symbol.strip()
    if any(c in symbol for c in "«»") or "Previous" in symbol or "Next" in symbol:
        return None
    if symbol.endswith("..L"):
        symbol = symbol.replace("..L", ".L")
    if ".A.L" in symbol:
        symbol = symbol.replace(".A.L", "-A.L")
    if ".B.L" in symbol:
        symbol = symbol.replace(".B.L", "-B.L")
    url = f"https://query1.finance.yahoo.com/v8/finance/chart/{symbol}"
    try:
        r = _session.get(url, params={"range": "1d", "interval": "1d"}, timeout=10)
        r.raise_for_status()
        meta = r.json()["chart"]["result"][0]["meta"]
        price = meta.get("regularMarketPrice")
        prev = meta.get("chartPreviousClose") or meta.get("previousClose")
        change = price - prev if price and prev else None
        change_pct = (change / prev * 100) if change and prev else None
        return {
            "symbol": symbol,
            "price": price,
            "change": change,
            "change_pct": change_pct,
            "volume": meta.get("regularMarketVolume"),
            "avg_volume": meta.get("averageDailyVolume3Month") or meta.get("averageDailyVolume10Day"),
            "name": meta.get("shortName") or meta.get("longName") or symbol,
        }
    except Exception as e:
        print(f"quote error {symbol}: {e}")
        return None

def fetch_quotes(symbols):
    results = []
    for sym in symbols:
        q = fetch_quote(sym)
        if q:
            results.append(q)
        time.sleep(0.15)
    return results

def fetch_news(symbol):
    try:
        url = f"https://query1.finance.yahoo.com/v1/finance/search"
        r = _session.get(url, params={"q": symbol, "newsCount": 5}, timeout=10)
        r.raise_for_status()
        items = r.json().get("news", [])[:5]
        result = []
        for it in items:
            result.append({
                "title": it.get("title"),
                "time": it.get("providerPublishTime"),
                "url": it.get("link") or it.get("url")
            })
        return result
    except Exception as e:
        print(f"news error {symbol}: {e}")
        return []

def fetch_ftse():
    try:
        r = _session.get(
            "https://query1.finance.yahoo.com/v8/finance/chart/%5EFTSE",
            params={"range": "5d", "interval": "1d"},
            timeout=10
        )
        r.raise_for_status()
        meta = r.json()["chart"]["result"][0]["meta"]
        prev_close = meta.get("chartPreviousClose") or meta.get("previousClose")
        current = meta.get("regularMarketPrice")

        now_uk = datetime.now(UK)
        open_t = now_uk.replace(hour=8, minute=0, second=0, microsecond=0)
        close_t = now_uk.replace(hour=16, minute=30, second=0, microsecond=0)
        is_weekday = now_uk.weekday() < 5
        is_open = is_weekday and open_t <= now_uk < close_t

        state = get_ftse_state()
        previous_was_up = state["previous_was_up"]

        if prev_close is not None and state["previous_close"] != prev_close:
            if state["previous_close"] is not None and current is not None:
                previous_was_up = current >= state["previous_close"]
                save_ftse_state(prev_close, previous_was_up)
            else:
                save_ftse_state(prev_close, None)

        return {
            "previous": prev_close,
            "current": current,
            "is_open": is_open,
            "previous_was_up": bool(previous_was_up) if previous_was_up is not None else None,
            "server_time": now_uk.isoformat()
        }
    except Exception as e:
        print("FTSE fetch error:", e)
        return {"previous": None, "current": None, "is_open": False, "previous_was_up": None, "server_time": None}

def _page1_fingerprint(url):
    """Fast fingerprint from the first page only."""
    try:
        r = _session.get(url, timeout=10)
        r.raise_for_status()
        soup = BeautifulSoup(r.text, "html.parser")
        epics = []
        for row in soup.select("table tr"):
            cells = row.find_all("td")
            if not cells:
                continue
            raw = cells[0].get_text(strip=True).strip()
            if raw and len(raw) <= 12 and " " not in raw and "«" not in raw:
                epics.append(raw)
        epics.sort()
        return hashlib.md5("|".join(epics).encode()).hexdigest()
    except Exception as e:
        print("Fingerprint failed:", e)
        return None

def refresh_constituents_if_needed():
    """Only full-scrape when the page-1 fingerprint has changed."""
    fp100 = _page1_fingerprint(INDEX_SOURCES["100"])
    fp250 = _page1_fingerprint(INDEX_SOURCES["250"])
    if not fp100 or not fp250:
        print("Fingerprint check failed — skipping refresh")
        return

    current_fp = f"{fp100}:{fp250}"
    stored_fp = get_constituents_fingerprint()

    if current_fp == stored_fp:
        print("Constituents unchanged — skipping full scrape")
        return

    print("Constituents changed — running full refresh")
    refresh_constituents()          # existing full scrape
    save_constituents_fingerprint(current_fp)

import sqlite3
from pathlib import Path

DB_PATH = "ape_stock.db"

def get_db():
    conn = sqlite3.connect(DB_PATH, timeout=30)
    conn.execute("PRAGMA journal_mode=WAL;")
    conn.execute("PRAGMA synchronous=NORMAL;")
    conn.execute("PRAGMA busy_timeout=30000;")
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        CREATE TABLE IF NOT EXISTS index_constituents (
            index_key TEXT,
            symbol TEXT,
            PRIMARY KEY (index_key, symbol)
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS index_quotes (
            symbol TEXT PRIMARY KEY,
            price REAL,
            change REAL,
            change_pct REAL,
            volume INTEGER,
            avg_volume INTEGER,
            name TEXT,
            last_update TEXT
        )
    """)
    cur.execute("""
        CREATE TABLE IF NOT EXISTS watchlist (
            symbol TEXT PRIMARY KEY,

            added_price REAL,
            news_count INTEGER DEFAULT 0,

            news_checked_at TEXT

    cur.execute("""

def get_ftse_state():
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT previous_close, previous_was_up FROM ftse_state WHERE id = 1")
    row = cur.fetchone()
    conn.close()
    return dict(row) if row else {"previous_close": None, "previous_was_up": None}

def save_ftse_state(previous_close, previous_was_up):
    conn = get_db()
    cur = conn.cursor()
    flag = None if previous_was_up is None else (1 if previous_was_up else 0)
    cur.execute("""
        UPDATE ftse_state SET previous_close = ?, previous_was_up = ? WHERE id = 1
    """, (previous_close, flag))
    conn.commit()
    conn.close()

def set_setting(key, value):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        INSERT INTO app_settings (key, value) VALUES (?, ?)

def close_holding_internal(hid, exit_price, reason):
    """Close a holding and archive it. Used by both the route and background."""
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM holdings WHERE id = ?", (hid,))
    h = cur.fetchone()
    if not h:
        conn.close()
        return None

    proceeds = h["shares"] * exit_price
    pnl_net = proceeds - h["notional"]

    cur.execute("""
        SELECT day_offset, price, pnl_net FROM holdings_marks
        WHERE holding_id = ? ORDER BY day_offset
    """, (hid,))
    path = [dict(r) for r in cur.fetchall()]

    cur.execute("""
        INSERT INTO holdings_history
        (symbol, entry_date, entry_price, exit_date, exit_price, notional, stamp_pct, shares,
         exit_reason, final_pnl_net, day_path, closed_at)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    """, (
        h["symbol"], h["entry_date"], h["entry_price"],
        datetime.now(UK).date().isoformat(), exit_price,
        h["notional"], h["stamp_pct"], h["shares"],
        reason, pnl_net, json.dumps(path),
        datetime.now(UK).isoformat()
    ))
    cur.execute("DELETE FROM holdings WHERE id = ?", (hid,))
    cur.execute("DELETE FROM holdings_marks WHERE holding_id = ?", (hid,))
    conn.commit()
    conn.close()
    return pnl_net

def refresh_live_holding_prices():
    """Update the highest day_offset mark with the live price. Retries briefly on database lock."""
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        SELECT h.id, h.symbol, h.shares, h.notional, h.entry_price,
               MAX(m.day_offset) as day_offset
        FROM holdings h
        JOIN holdings_marks m ON m.holding_id = h.id
        WHERE h.status = 'open'
        GROUP BY h.id
    """)
    holdings = cur.fetchall()
    conn.close()

    if not holdings:
        return

    today = datetime.now(UK).date().isoformat()
    for h in holdings:
        price = None
        try:
            q = fetch_quote(h["symbol"])
            price = q.get("price") if q else None
        except Exception:
            continue
        if price is None:
            continue

        pnl_net = (h["shares"] * price) - h["notional"] if h["shares"] else None
        day_offset = h["day_offset"] or 0

        for attempt in range(3):
            try:
                conn = get_db()
                cur = conn.cursor()
                cur.execute("""
                    UPDATE holdings_marks
                    SET price = ?, pnl_net = ?, mark_date = ?
                    WHERE holding_id = ? AND day_offset = ?
                """, (price, pnl_net, today, h["id"], day_offset))
                conn.commit()
                conn.close()
                break
            except sqlite3.OperationalError as e:
                if "locked" in str(e).lower() and attempt < 2:
                    time.sleep(0.15)
                    continue
                try:
                    conn.close()
                except Exception:
                    pass
                break
            except Exception:
                try:
                    conn.close()
                except Exception:
                    pass
                break

def open_holding():
    data = request.get_json(silent=True) or {}
    symbol = data.get("symbol")
    price = data.get("price")
    print(f"[holdings/open] request for {symbol}", flush=True)
    if not symbol or not price:
        return jsonify({"error": "symbol and price required"}), 400

    conn = get_db()
    cur = conn.cursor()

    cur.execute("SELECT id FROM holdings WHERE symbol = ? AND status = 'open'", (symbol,))
    existing = cur.fetchone()
    if existing:
        print(f"[holdings/open] REJECTED duplicate open for {symbol} (id={existing['id']})", flush=True)
        conn.close()
        return jsonify({"error": "already open", "symbol": symbol}), 409

    notional = float(data.get("notional") or 500)
    stamp_pct = float(data.get("stamp_pct") or 0.5)
    cost = notional * (stamp_pct / 100.0)
    invest = notional - cost
    shares = invest / float(price)

    now = datetime.now(UK).isoformat()
    entry_date = datetime.now(UK).date().isoformat()
    cur.execute("""
        INSERT INTO holdings (symbol, entry_date, entry_price, notional, stamp_pct, shares, status, created)
        VALUES (?, ?, ?, ?, ?, ?, 'open', ?)
    """, (symbol, entry_date, price, notional, stamp_pct, shares, now))
    holding_id = cur.lastrowid

    cur.execute("""
        INSERT INTO holdings_marks (holding_id, day_offset, mark_date, price, pnl_net)
        VALUES (?, 0, ?, ?, 0)
    """, (holding_id, entry_date, price))
    conn.commit()
    conn.close()
    print(f"[holdings/open] created id={holding_id} for {symbol}", flush=True)
    return jsonify({"ok": True, "id": holding_id})

def get_index_data(index_key, direction, limit=None, min_change=None, max_change=None):
    conn = get_db()
    cur = conn.cursor()
    if index_key == "ALL":
        cur.execute("SELECT symbol FROM index_constituents")
    else:
        cur.execute("SELECT symbol FROM index_constituents WHERE index_key = ?", (index_key,))
    symbols = [row[0] for row in cur.fetchall()]
    if not symbols:
        conn.close()
        return []

    placeholders = ",".join("?" * len(symbols))
    order = "DESC" if direction == "gainers" else "ASC"

    where_extra = ""
    params = symbols[:]
    if min_change is not None and direction == "gainers":
        where_extra += " AND change_pct >= ?"
        params.append(min_change)
    if max_change is not None and direction == "losers":
        where_extra += " AND change_pct <= ?"
        params.append(max_change)

    limit_sql = ""
    if limit is not None:
        limit_sql = "LIMIT ?"
        params.append(limit)

    cur.execute(f"""
        SELECT * FROM index_quotes
        WHERE symbol IN ({placeholders})
          AND change_pct IS NOT NULL
          {where_extra}
        ORDER BY change_pct {order}
        {limit_sql}
    """, params)
    rows = [dict(row) for row in cur.fetchall()]
    conn.close()
    return rows

@bp.route("/rsi/<symbol>")
def rsi(symbol):
    """Return RSI(7) from daily closes."""
    conn = get_db()
    cur = conn.cursor()
    cur.execute("""
        SELECT close FROM daily_closes
        WHERE symbol = ? AND close IS NOT NULL
        ORDER BY date DESC LIMIT 30
    """, (symbol,))
    closes = [r[0] for r in cur.fetchall()][::-1]  # oldest first
    conn.close()
    if len(closes) < 8:
        return jsonify({"rsi7": None})

    def calc_rsi(prices, period=7):
        if len(prices) < period + 1:
            return None
        deltas = [prices[i] - prices[i-1] for i in range(1, len(prices))]
        gains = [d if d > 0 else 0 for d in deltas]
        losses = [-d if d < 0 else 0 for d in deltas]
        avg_gain = sum(gains[:period]) / period
        avg_loss = sum(losses[:period]) / period
        for i in range(period, len(deltas)):
            avg_gain = (avg_gain * (period-1) + gains[i]) / period
            avg_loss = (avg_loss * (period-1) + losses[i]) / period
        if avg_loss == 0:
            return 100.0
        rs = avg_gain / avg_loss
        return 100 - (100 / (1 + rs))

    return jsonify({"rsi7": round(calc_rsi(closes, 7), 1)})

from flask import Blueprint, request, jsonify

@bp.route("/scripts/run", methods=["POST"])
def scripts_run():
    data = request.get_json(silent=True) or {}
    name = data.get("script", "").strip()
    if not name or not name.endswith(".py") or "/" in name or "\\" in name:
        return jsonify({"error": "Invalid script name"}), 400
    script_path = SCRIPTS_DIR / name
    if not script_path.exists():
        return jsonify({"error": "Script not found"}), 404
    try:
        result = subprocess.run(
            [sys.executable, str(script_path)],
            capture_output=True,
            text=True,
            timeout=180,
            cwd=str(SCRIPTS_DIR),
            encoding="utf-8",
            errors="replace"
        )
        return jsonify({
            "stdout": result.stdout,
            "stderr": result.stderr,
            "returncode": result.returncode
        })
    except subprocess.TimeoutExpired:
        return jsonify({"error": "Script timed out"}), 500
    except Exception as e:
        return jsonify({"error": str(e)}), 500

function formatNumber(value, type = "price") {
    if (value == null || isNaN(value)) return "";
    const abs = Math.abs(value);
    if (type === "volume") return value.toLocaleString();
    let decimals = 2;
    if (abs === 0) decimals = 2;
    else if (abs < 0.0001) decimals = 6;
    else if (abs < 0.01) decimals = 4;
    else if (abs < 1) decimals = 3;
    return value.toLocaleString(undefined, {
        minimumFractionDigits: decimals,
        maximumFractionDigits: decimals
    });
}

function makeTableSortable(table, options = {}) {
    if (!table) return;
    const getVal = options.getValue || ((cell) => cell.textContent.trim());
    let sortKey = null;
    let sortDir = 1;

    table.querySelectorAll("th[data-key]").forEach(th => {
        th.style.cursor = "pointer";
        th.addEventListener("click", () => {
            const key = th.dataset.key;
            if (sortKey === key) sortDir *= -1;
            else { sortKey = key; sortDir = 1; }

            const tbody = table.querySelector("tbody");
            const rows = [...tbody.querySelectorAll("tr")];
            const colIndex = th.cellIndex;

            rows.sort((a, b) => {
                // Skip details rows (they have fewer cells)
                if (a.classList.contains("details-row") || b.classList.contains("details-row")) return 0;

                const cellA = a.children[colIndex];
                const cellB = b.children[colIndex];
                if (!cellA || !cellB) return 0;

                let va = getVal(cellA);
                let vb = getVal(cellB);
                const na = parseFloat(String(va).replace(/[^0-9.-]/g, ""));
                const nb = parseFloat(String(vb).replace(/[^0-9.-]/g, ""));
                if (!isNaN(na) && !isNaN(nb)) return (na - nb) * sortDir;
                return String(va).localeCompare(String(vb)) * sortDir;
            });

            // Keep details rows attached to their main row
            const details = [...tbody.querySelectorAll("tr.details-row")];
            rows.forEach(r => {
                if (r.classList.contains("details-row")) return; // already handled
                tbody.appendChild(r);
                const sym = r.dataset.symbol;
                const det = details.find(d => d.previousElementSibling === r || d.dataset.symbol === sym);
                if (det) tbody.appendChild(det);
            });

            // Update sort arrows WITHOUT destroying child elements
            table.querySelectorAll("th[data-key]").forEach(h => {
                // Remove only the arrow text node, leave any inputs alone
                [...h.childNodes].forEach(node => {
                    if (node.nodeType === Node.TEXT_NODE) {
                        node.textContent = node.textContent.replace(/ ↑| ↓/, "");
                    }
                });
            });
            // Append arrow as a text node
            th.appendChild(document.createTextNode(sortDir === 1 ? " ↑" : " ↓"));

            if (options.onSort) options.onSort(key, sortDir);
        });
    });
}

function drawSparkline(canvas, values, style = "line") {
    if (!canvas || !values || values.length < 2) return;
    const ctx = canvas.getContext("2d");
    const w = canvas.width = canvas.offsetWidth || 80;
    const h = canvas.height = canvas.offsetHeight || 24;
    ctx.clearRect(0, 0, w, h);
    const min = Math.min(...values);
    const max = Math.max(...values);
    const range = max - min || 1;
    const step = w / (values.length - 1);

    const points = values.map((v, i) => ({
        x: i * step,
        y: h - ((v - min) / range) * (h - 4) - 2
    }));

    ctx.lineWidth = 1.5;
    if (style === "bars") {
        points.forEach((p, i) => {
            const up = i === 0 || values[i] >= values[i - 1];
            ctx.fillStyle = up ? "#4ade80" : "#f87171";
            const barH = Math.abs(p.y - h / 2);
            ctx.fillRect(p.x - 1, Math.min(p.y, h / 2), 2, barH || 1);
        });
        return;
    }
    if (style === "winloss") {
        points.forEach((p, i) => {
            if (i === 0) return;
            ctx.fillStyle = values[i] >= values[i - 1] ? "#4ade80" : "#f87171";
            ctx.fillRect(p.x - 1, h / 2 - 3, 2, 6);
        });
        return;
    }

    ctx.beginPath();
    ctx.moveTo(points[0].x, points[0].y);
    points.forEach(p => ctx.lineTo(p.x, p.y));
    if (style === "gradient") {
        const grad = ctx.createLinearGradient(0, 0, w, 0);
        grad.addColorStop(0, "#4ade80");
        grad.addColorStop(1, "#f87171");
        ctx.strokeStyle = grad;
    } else {
        ctx.strokeStyle = values[values.length - 1] >= values[0] ? "#4ade80" : "#f87171";
    }
    ctx.stroke();

    if (style === "area") {
        ctx.lineTo(points[points.length - 1].x, h);
        ctx.lineTo(points[0].x, h);
        ctx.closePath();
        ctx.fillStyle = "rgba(74, 222, 128, 0.15)";
        ctx.fill();
    }
}

function nextOpenDescription(nowIso) {
    if (!nowIso) return null;
    const ukNow = new Date(nowIso);
    const day = ukNow.getDay();
    const openToday = new Date(ukNow);
    openToday.setHours(8, 0, 0, 0);
    const closeToday = new Date(ukNow);
    closeToday.setHours(16, 30, 0, 0);

    let target = new Date(openToday);
    let label = "";

    if (day === 0) {                       // Sunday
        label = "Opens [tomorrow in]:";
        target.setDate(target.getDate() + 1);
    } else if (day === 6) {                // Saturday
        label = "Opens [on Monday in]:";
        target.setDate(target.getDate() + 2);
    } else if (ukNow >= closeToday) {      // after close
        label = "Opens [tomorrow in]:";
        target.setDate(target.getDate() + 1);
        if (target.getDay() === 6) {
            label = "Opens [on Monday in]:";
            target.setDate(target.getDate() + 2);
        } else if (target.getDay() === 0) {
            label = "Opens [on Monday in]:";
            target.setDate(target.getDate() + 1);
        }
    } else if (ukNow < openToday) {        // before open
        label = "Opens [today in]:";
    } else {
        return null;                       // market is open
    }

    // Safety: if target is already past, push to next weekday open
    if (target <= ukNow) {
        target.setDate(target.getDate() + 1);
        while (target.getDay() === 0 || target.getDay() === 6) {
            target.setDate(target.getDate() + 1);
        }
        label = "Opens [next in]:";
    }

    return { label, target };
}

function startFtseCountdown(target, labelEl) {
    if (ftseCountdownTimer) clearInterval(ftseCountdownTimer);
    function tick() {
        const diff = target - new Date();
        if (diff <= 0) {
            labelEl.textContent = "Open";
            clearInterval(ftseCountdownTimer);
            return;
        }
        const h = Math.floor(diff / 3600000);
        const m = Math.floor((diff % 3600000) / 60000);
        const s = Math.floor((diff % 60000) / 1000);
        labelEl.textContent = `${String(h).padStart(2,"0")}:${String(m).padStart(2,"0")}:${String(s).padStart(2,"0")}`;
    }
    tick();
    ftseCountdownTimer = setInterval(tick, 1000);
}

async function updateCacheStatus() {
    try {
        const res = await fetch("/status");
        const data = await res.json();
        const last = data.last_update ? new Date(data.last_update).toLocaleTimeString() : "never";
        document.getElementById("cache-status").textContent =
            `Cache: ${data.symbols || 0} symbols · last update ${last}`;

        const ftse = data.ftse || {};
        const el = document.getElementById("ftse-status");
        const prevText = ftse.previous != null ? formatNumber(ftse.previous, "price") : "—";
        const prevCls = ftse.previous_was_up === true ? "positive" :
                        ftse.previous_was_up === false ? "negative" : "";

        if (ftse.is_open && ftse.current != null) {
            const currText = formatNumber(ftse.current, "price");
            const currCls = ftse.previous != null ? (ftse.current >= ftse.previous ? "positive" : "negative") : "";

            let changeText = "—";
            let changePctText = "—";
            let changeCls = "";

            if (ftse.previous != null && ftse.current != null) {
                const change = ftse.current - ftse.previous;
                const changePct = (change / ftse.previous) * 100;
                changeText = (change >= 0 ? "+" : "") + formatNumber(change, "price");
                changePctText = (changePct >= 0 ? "+" : "") + changePct.toFixed(2) + "%";
                changeCls = change >= 0 ? "positive" : "negative";
            }

            el.innerHTML = `
                FTSE Previous: <span class="${prevCls}">${prevText}</span>
                &nbsp;&nbsp;FTSE Current: <span id="ftse-current" class="${currCls}">${currText}</span>
                &nbsp;&nbsp;Change: <span class="${changeCls}">${changeText}</span>
                &nbsp;&nbsp;Change %: <span class="${changeCls}">${changePctText}</span>
            `;

            if (lastFtseCurrent != null && ftse.current !== lastFtseCurrent) {
                const cell = document.getElementById("ftse-current");
                if (cell) triggerFlash(cell, ftse.current > lastFtseCurrent);
            }
            lastFtseCurrent = ftse.current;
            if (ftseCountdownTimer) clearInterval(ftseCountdownTimer);
        } else {
            const info = nextOpenDescription(ftse.server_time);
            const prevCls = ftse.previous_was_up === true ? "positive" :
                            ftse.previous_was_up === false ? "negative" : "";
            const prevText = ftse.previous != null ? formatNumber(ftse.previous, "price") : "—";

            el.innerHTML = `
                FTSE Previous: <span class="${prevCls}">${prevText}</span>
                &nbsp;&nbsp;FTSE Current: Closed
                ${info ? ` - <span id="ftse-open-label"></span> <span id="ftse-countdown"></span>` : ""}
            `;

            if (info) {
                document.getElementById("ftse-open-label").textContent = info.label;
                startFtseCountdown(info.target, document.getElementById("ftse-countdown"));
            }
        }
    } catch (e) {}
}

// Re-apply a saved sort only if the visual order has actually changed.
// Avoids unnecessary DOM moves (and therefore flicker) on price ticks.
function reapplySortIfChanged(table, key, dir) {
    if (!table || !key) return;
    const th = table.querySelector(`th[data-key="${key}"]`);
    if (!th) return;

    const tbody = table.querySelector("tbody");
    if (!tbody) return;
    const rows = [...tbody.querySelectorAll("tr")];
    if (rows.length < 2) return;

    const colIndex = th.cellIndex;

    const sorted = [...rows].sort((a, b) => {
        let va = a.children[colIndex].textContent.trim();
        let vb = b.children[colIndex].textContent.trim();
        const na = parseFloat(String(va).replace(/[^0-9.-]/g, ""));
        const nb = parseFloat(String(vb).replace(/[^0-9.-]/g, ""));
        if (!isNaN(na) && !isNaN(nb)) return (na - nb) * dir;
        return String(va).localeCompare(String(vb)) * dir;
    });

    // Already in the correct order?
    let same = true;
    for (let i = 0; i < rows.length; i++) {
        if (rows[i] !== sorted[i]) {
            same = false;
            break;
        }
    }
    if (same) return;

    // Order changed — move rows
    sorted.forEach(r => tbody.appendChild(r));

    // Update sort indicator
    table.querySelectorAll("th[data-key]").forEach(h => {
        [...h.childNodes].forEach(node => {
            if (node.nodeType === Node.TEXT_NODE) {
                node.textContent = node.textContent.replace(/ ↑| ↓/, "");
            }
        });
    });
    th.appendChild(document.createTextNode(dir === 1 ? " ↑" : " ↓"));
}

    async render(files) {
        const container = document.getElementById("results");
        let listHtml = "";

        if (!files.length) {
            listHtml = "<p>No scripts found in /scripts.</p>";
        } else {
            // Fetch DESC / RUN for each script
            const meta = await Promise.all(files.map(async f => {
                try {
                    const r = await fetch(`/scripts/meta?name=${encodeURIComponent(f)}`);
                    return await r.json();
                } catch {
                    return { name: f, desc: "", run: "" };
                }
            }));

            listHtml = `<ul class="script-list">`;
            meta.forEach(m => {
                const badge = m.run === "VENV"
                    ? `<span class="run-badge venv">VENV</span>`
                    : `<span class="run-badge tool2">Tool2</span>`;
                const desc = m.desc ? `<div class="script-desc">${m.desc}</div>` : "";
                listHtml += `<li>
                    <label>
                        <input type="radio" name="script-choice" value="${m.name}">
                        <strong>${m.name}</strong> ${badge}
                        ${desc}
                    </label>
                </li>`;
            });
            listHtml += `</ul>`;
        }

        container.innerHTML = `
            <div class="region-bar">
                <h3>Ad-Hoc Scripts</h3>
                <div class="right-controls">
                    <button class="mini-btn" id="script-cancel">Cancel</button>
                    <button class="mini-btn" id="script-run">Run</button>
                </div>
            </div>
            <div id="script-list-area">${listHtml}</div>
            <pre id="script-output" class="script-output"></pre>`;

        document.getElementById("script-cancel").addEventListener("click", () => {
            const dash = document.querySelector('[data-tool="dashboard"]') ||
                         document.querySelector('a[href="/"]') ||
                         document.querySelector('button[data-tool="1"]');
            if (dash) dash.click();
            else if (window.Dashboard) window.Dashboard.show();
        });

        document.getElementById("script-run").addEventListener("click", async () => {
            const selected = document.querySelector('input[name="script-choice"]:checked');
            if (!selected) {
                alert("Select a script first");
                return;
            }

            const output = document.getElementById("script-output");
            const name = selected.value;

            // Check run mode first
            try {
                const metaRes = await fetch(`/scripts/meta?name=${encodeURIComponent(name)}`);
                const meta = await metaRes.json();
                if (meta.run && meta.run.toUpperCase() === "VENV") {
                    output.textContent = `This script must be run externally.\n\n` +
                                        `Open a terminal, activate the venv, and run:\n` +
                                        `python scripts/${name}`;
                    return;
                }
            } catch (e) {
                // if meta fails, continue and let the backend decide
            }

            output.textContent = "Running…";
            try {
                const res = await fetch("/scripts/run", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({ script: name })
                });
                const data = await res.json();
                if (data.error) {
                    output.textContent = "Error: " + data.error;
                } else {
                    output.textContent = (data.stdout || "") + (data.stderr ? "\n--- stderr ---\n" + data.stderr : "");
                }
            } catch (e) {
                output.textContent = "Request failed: " + e;
            }
        });
    }
};

                drawSparkline(canvas, prices, "line");

                drawSparkline(canvas, [1, 1], "line");

        });

            <th data-key="volume">Volume</th>

        for (const r of toShow) {

            }
            if (volRatioVal && volRatioVal >= volRatioThreshold) {

                reasons.push(`Vol/Avg ≥ ${volRatioThreshold}× (${volRatioVal.toFixed(1)}×)`);

                reasons.push(`RSI(7) ≤ 30 (${r.rsi7.toFixed(1)}) on faller`);
            }

            const hitPct = Math.round((points / possible) * 100);
            const hitCls = hitPct >= 75 ? "positive" : hitPct >= 50 ? "" : "negative";
            const hitTitle = reasons.length ? reasons.join("\n") : "No rules matched";
            const hitText = `${hitPct}%`;
            const hitThreshold = parseFloat(localStorage.getItem("hitThreshold") || "50");
            const highlight = hitPct >= hitThreshold ? 'style="background:rgba(74,222,128,0.12)"' : "";

            html += `<tr class="main-row" data-symbol="${r.symbol}" ${highlight}>

                if (r.volume && r.avg_volume) {
                    volRatio = (r.volume / r.avg_volume).toFixed(1) + "×";
                } else if (r.volume) {

                }

                        <strong>Sector:</strong> ${sector} &nbsp;·&nbsp;
                        <strong>Industry:</strong> ${industry}
                    </td>
                </tr>`;
            }
        }
        html += "</tbody></table>";
        return html;
    },

                (symbol, date, open, high, low, close, volume)
                VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (sym, date_str, o, h, l, c, v))

from pathlib import Path

ROOT = Path(__file__).parent.parent
OUT = ROOT / "handover-prompt.txt"

# Everything we want (explicit so venv, __pycache__, db, etc. are excluded)
INCLUDE_DIRS = ["templates", "static", "scripts"]
INCLUDE_ROOT_FILES = [
    "app.py",
    "background.py",
    "data.py",
    "db.py",
    "holdings.py",
    "routes.py",
    "scripts_runner.py",
    "requirements.txt",
    "run.bat",
    "set-env.bat",
]

def should_include(path: Path) -> bool:
    if path.name.startswith(".") or path.name == "__pycache__":
        return False
    if path.suffix in {".pyc", ".db", ".txt"} and path.name != "requirements.txt":
        return False
    return True

def build_tree():
    lines = ["ape-stock/"]
    # root files
    for name in sorted(INCLUDE_ROOT_FILES):
        p = ROOT / name
        if p.exists():
            lines.append(f"├── {name}")
    # directories
    for dname in INCLUDE_DIRS:
        d = ROOT / dname
        if not d.exists():
            continue
        lines.append(f"├── {dname}/")
        for f in sorted(d.rglob("*")):
            if f.is_file() and should_include(f):
                rel = f.relative_to(ROOT)
                indent = "│   " * (len(rel.parts) - 1)
                lines.append(f"{indent}├── {rel.name}")
    return "\n".join(lines)

def file_blocks():
    blocks = []
    # root files
    for name in INCLUDE_ROOT_FILES:
        p = ROOT / name
        if p.exists() and should_include(p):
            content = p.read_text(encoding="utf-8", errors="replace")
            blocks.append(f"===== FILE: {name} =====\n{content}\n")
    # directory files
    for dname in INCLUDE_DIRS:
        d = ROOT / dname
        if not d.exists():
            continue
        for f in sorted(d.rglob("*")):
            if f.is_file() and should_include(f):
                rel = f.relative_to(ROOT).as_posix()
                content = f.read_text(encoding="utf-8", errors="replace")
                blocks.append(f"===== FILE: {rel} =====\n{content}\n")
    return "\n".join(blocks)

import sqlite3

    FROM daily_closes
    WHERE close IS NOT NULL
    ORDER BY symbol, date
""")

for r in cur.fetchall():
    by_sym[r["symbol"]].append((r["date"], r["close"]))

sym_dates = {}
for sym, series in by_sym.items():
    sym_dates[sym] = {d: c for d, c in series}

print(f"Found {len(exdivs)} symbols with an ex-div date")

# Collect returns for T-5 ... T-1 and a simple control (T-10)
days_before = [5, 4, 3, 2, 1]
results = {d: [] for d in days_before}
control = []
yield_buckets = {"low": [], "mid": [], "high": []}  # <1%, 1-3%, >3%

    sym = row["symbol"]
    ex_str = row["ex_dividend_date"]
    dy = row["dividend_yield"] or 0
    if sym not in sym_dates:
        continue
    try:
        ex_dt = datetime.fromisoformat(ex_str[:10]).date()
    except Exception:
        continue

    dates = sym_dates[sym]
    # Find trading days around the ex-date
    # We need the close on the day before each offset
    sorted_dates = sorted(dates.keys())
    # Locate the last trading day strictly before ex_dt
    pre_dates = [d for d in sorted_dates if d < ex_dt.isoformat()]
    if len(pre_dates) < 6:
        continue

    # T-1 is the last pre_date, T-2 the one before, etc.
    for offset in days_before:
        if len(pre_dates) >= offset + 1:
            d_now = pre_dates[-offset]
            d_prev = pre_dates[-offset - 1]
            c_now = dates[d_now]
            c_prev = dates[d_prev]
            if c_prev:
                ret = (c_now - c_prev) / c_prev * 100
                results[offset].append(ret)
                # bucket by yield
                if dy < 1:
                    yield_buckets["low"].append((offset, ret))
                elif dy < 3:
                    yield_buckets["mid"].append((offset, ret))
                else:
                    yield_buckets["high"].append((offset, ret))

    if len(pre_dates) >= 11:
        d_now = pre_dates[-10]
        d_prev = pre_dates[-11]
        c_now = dates[d_now]
        c_prev = dates[d_prev]

    wins = sum(1 for v in vals if v > 0) / len(vals) * 100

def calc_rsi(closes, period=7):
    if len(closes) < period + 1:
        return None
    deltas = [closes[i] - closes[i-1] for i in range(1, len(closes))]
    gains = [d if d > 0 else 0 for d in deltas]
    losses = [-d if d < 0 else 0 for d in deltas]
    avg_gain = sum(gains[:period]) / period
    avg_loss = sum(losses[:period]) / period

        avg_loss = (avg_loss * (period - 1) + losses[i]) / period

        continue

    for i, date in enumerate(sorted_dates):
        bars = sorted(dates[date])
        if len(bars) < 8:
            continue
        prices = [b[1] for b in bars]
        day_open = prices[0]

        # Find first bar that crosses each drop threshold from the open
        for thresh in DROP_PCTS:
            entry_price = None
            for p in prices:
                drop = (p - day_open) / day_open * 100 if day_open else 0
                if drop <= -thresh:
                    entry_price = p          # buy at the close of this 5-min bar
                    break
            if entry_price is None:
                continue

            results[thresh]["signals"] += 1
            day_close = prices[-1]

            # same-day net from realistic entry
            same_gross = (day_close - entry_price) / entry_price * 100
            results[thresh]["same_day"].append(same_gross - STAMP)

            # next-day close
            if i + 1 < len(sorted_dates):
                next_bars = sorted(dates[sorted_dates[i + 1]])
                if next_bars:
                    next_close = next_bars[-1][1]
                    next_gross = (next_close - entry_price) / entry_price * 100
                    results[thresh]["next_day"].append(next_gross - STAMP)

      f"{'NextDay Avg%':>13} {'NextDay Med%':>13}")

    r = results[thresh]
    def avg(xs): return sum(xs) / len(xs) if xs else 0
    def med(xs):
        if not xs: return 0
        s = sorted(xs)
        return s[len(s) // 2]
    print(f"{thresh:7} {r['signals']:8} {avg(r['same_day']):13.2f} {med(r['same_day']):13.2f} "

def build_screener(as_of_date):
    candidates = []
    for sym, series in by_sym.items():
        closes = [c for d, c in series if d <= as_of_date]
        n = len(closes)
        if n < 200:
            continue
        ret_3m = (closes[-1] / closes[-63] - 1) * 100 if n > 63 and closes[-63] else None
        if ret_3m is None or ret_3m <= -15:

    positions = {}

    stamp = 0.005

    for i in range(start_idx, len(sorted_dates)):

    ret_3m = (closes[-1] / closes[-63] - 1) * 100 if n > 63 and closes[-63] else None
    ret_6m = (closes[-1] / closes[-126] - 1) * 100 if n > 126 and closes[-126] else None

    lookback = min(252, n)
    high_52w = max(closes[-lookback:])
    dist_52w = (closes[-1] / high_52w - 1) * 100 if high_52w else None

    def sma(p):
        return sum(closes[-p:]) / p if n >= p else None

    ma50 = sma(50)
    ma200 = sma(200)
    above_50 = 1 if ma50 and closes[-1] > ma50 else 0
    pct_50 = (closes[-1] / ma50 - 1) * 100 if ma50 else None
    above_200 = 1 if ma200 and closes[-1] > ma200 else 0
    pct_200 = (closes[-1] / ma200 - 1) * 100 if ma200 else None

    vol_20d = None
    if n >= 21:
        rets = [(closes[i] / closes[i-1] - 1) for i in range(n-20, n) if closes[i-1]]
        if rets:
            mean = sum(rets) / len(rets)
            var = sum((r - mean)**2 for r in rets) / len(rets)
            vol_20d = math.sqrt(var) * 100

    ret_1d = (closes[-1] / closes[-2] - 1) * 100 if n >= 2 and closes[-2] else None
    ret_5d = (closes[-1] / closes[-6] - 1) * 100 if n >= 6 and closes[-6] else None

    sector = sectors.get(sym)
    if sector and ret_1d is not None:
        sector_1d[sector].append(ret_1d)
    if sector and ret_5d is not None:
        sector_5d[sector].append(ret_5d)

    updates.append((
        ret_3m, ret_6m, dist_52w,
        above_50, pct_50, above_200, pct_200,
        vol_20d, None, None,
        sym, ret_1d, ret_5d, sector
    ))

avg_1d = {s: sum(v)/len(v) for s, v in sector_1d.items() if v}
avg_5d = {s: sum(v)/len(v) for s, v in sector_5d.items() if v}

print(f"Writing {len(updates)} rows...")
batch = []
for u in updates:
    ret_3m, ret_6m, dist_52w, above_50, pct_50, above_200, pct_200, vol_20d, _, _, sym, ret_1d, ret_5d, sector = u
    rel_1d = (ret_1d - avg_1d[sector]) if sector and ret_1d is not None and sector in avg_1d else None
    rel_5d = (ret_5d - avg_5d[sector]) if sector and ret_5d is not None and sector in avg_5d else None
    batch.append((
        ret_3m, ret_6m, dist_52w,
        above_50, pct_50, above_200, pct_200,
        vol_20d, rel_1d, rel_5d, sym
    ))

cur.executemany("""
    UPDATE index_quotes
    SET ret_3m=?, ret_6m=?, dist_52w_high=?,
        above_50ma=?, pct_from_50ma=?,
        above_200ma=?, pct_from_200ma=?,
        vol_20d=?, sector_rel_1d=?, sector_rel_5d=?
    WHERE symbol=?
""", batch)

# Record timestamp
now = datetime.now(UK).isoformat()
cur.execute("""
    INSERT INTO app_settings (key, value)
    VALUES ('screener_metrics_updated', ?)
    ON CONFLICT(key) DO UPDATE SET value = excluded.value
""", (now,))

conn.commit()
conn.close()
print("Done. Screener metrics updated.")

"""
Add sector & industry to index_quotes for all FTSE constituents.
"""
import sqlite3
import time
from pathlib import Path
import yfinance as yf
import warnings
warnings.filterwarnings("ignore", category=Warning)

db_path = Path(__file__).parent.parent / "ape_stock.db"
conn = sqlite3.connect(db_path, timeout=30)
cur = conn.cursor()

# Add columns if they don't exist
for col in ("sector", "industry"):
    try:
        cur.execute(f"ALTER TABLE index_quotes ADD COLUMN {col} TEXT")
        print(f"Added column: {col}")
    except sqlite3.OperationalError:
        pass  # already exists
conn.commit()

# Get all symbols
cur.execute("SELECT DISTINCT symbol FROM index_constituents")
symbols = [r[0] for r in cur.fetchall()]
print(f"Updating sector/industry for {len(symbols)} symbols…")

updated = 0
for i, sym in enumerate(symbols, 1):
    try:
        info = yf.Ticker(sym).info
        sector = info.get("sector") or info.get("sectorDisp")
        industry = info.get("industry") or info.get("industryDisp")
        if sector or industry:
            cur.execute("""
                UPDATE index_quotes
                SET sector = ?, industry = ?
                WHERE symbol = ?
            """, (sector, industry, sym))
            if cur.rowcount:
                updated += 1
        if i % 25 == 0:
            print(f"  …{i}/{len(symbols)}", flush=True)
            conn.commit()
        time.sleep(0.25)
    except Exception as e:
        print(f"  {sym}: {e}")

conn.commit()
conn.close()
print(f"Done. Updated {updated} symbols.")
